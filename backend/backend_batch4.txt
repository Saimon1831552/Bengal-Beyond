/* ============================================================================
 * 1. UNIFIED TRANSACTIONS LEDGER  (/api/admin/transactions)
 * Combines `payments` (+) and `refunds` (-) into a single filterable ledger
 * ========================================================================== */
router.get(
  "/transactions",
  authorize("payments.read"),
  asyncHandler(async (req, res) => {
    const type = req.query.type || ""; // 'payment' | 'refund' | ''
    const status = req.query.status || "";
    const q = (req.query.q || "").trim();

    const parts = [];
    const params = [];

    if (!type || type === "payment") {
      const pWhere = ["1=1"];
      if (status) { pWhere.push("p.status = ?"); params.push(status); }
      if (q) {
        pWhere.push("(b.booking_reference LIKE ? OR b.customer_name LIKE ? OR p.provider_payment_id LIKE ?)");
        params.push(`%${q}%`, `%${q}%`, `%${q}%`);
      }
      parts.push(`
        SELECT 'payment' AS entry_type,
               p.id AS entry_id,
               p.booking_id,
               b.booking_reference,
               b.customer_name,
               b.customer_email,
               p.provider,
               p.provider_payment_id AS reference_id,
               p.amount,
               p.currency_code,
               p.status,
               COALESCE(p.verified_at, p.created_at) AS occurred_at
          FROM payments p
          JOIN bookings b ON b.id = p.booking_id
         WHERE ${pWhere.join(" AND ")}
      `);
    }

    if (!type || type === "refund") {
      const rWhere = ["1=1"];
      if (status) { rWhere.push("r.status = ?"); params.push(status); }
      if (q) {
        rWhere.push("(b.booking_reference LIKE ? OR b.customer_name LIKE ? OR r.provider_refund_id LIKE ?)");
        params.push(`%${q}%`, `%${q}%`, `%${q}%`);
      }
      parts.push(`
        SELECT 'refund' AS entry_type,
               r.id AS entry_id,
               p.booking_id,
               b.booking_reference,
               b.customer_name,
               b.customer_email,
               p.provider,
               r.provider_refund_id AS reference_id,
               r.amount,
               p.currency_code,
               r.status,
               COALESCE(r.processed_at, r.created_at) AS occurred_at
          FROM refunds r
          JOIN payments p ON p.id = r.payment_id
          JOIN bookings b ON b.id = p.booking_id
         WHERE ${rWhere.join(" AND ")}
      `);
    }

    const sql = `${parts.join(" UNION ALL ")} ORDER BY occurred_at DESC LIMIT 200`;
    const [rows] = await pool.execute(sql, params);
    return sendSuccess(res, rows);
  })
);

router.get(
  "/reports",
  authorize("payments.read"),
  asyncHandler(async (req, res) => {
    const from = req.query.from || null;
    const to = req.query.to || null;

    const pWhere = ["p.status = 'succeeded'"];
    const pParams = [];
    if (from) { pWhere.push("DATE(COALESCE(p.verified_at, p.created_at)) >= ?"); pParams.push(from); }
    if (to) { pWhere.push("DATE(COALESCE(p.verified_at, p.created_at)) <= ?"); pParams.push(to); }

    const [[grossRow]] = await pool.execute(
      `SELECT COALESCE(SUM(p.amount), 0) AS gross_revenue,
              COUNT(*) AS payment_count
         FROM payments p
        WHERE ${pWhere.join(" AND ")}`,
      pParams
    );

    const rWhere = ["r.status = 'succeeded'"];
    const rParams = [];
    if (from) { rWhere.push("DATE(COALESCE(r.processed_at, r.created_at)) >= ?"); rParams.push(from); }
    if (to) { rWhere.push("DATE(COALESCE(r.processed_at, r.created_at)) <= ?"); rParams.push(to); }

    const [[refundRow]] = await pool.execute(
      `SELECT COALESCE(SUM(r.amount), 0) AS total_refunds,
              COUNT(*) AS refund_count
         FROM refunds r
        WHERE ${rWhere.join(" AND ")}`,
      rParams
    );

    const [[bookingStats]] = await pool.execute(
      `SELECT COUNT(*) AS total_bookings,
              SUM(CASE WHEN booking_status IN ('confirmed', 'completed') THEN 1 ELSE 0 END) AS confirmed_bookings,
              SUM(CASE WHEN booking_status = 'cancelled' THEN 1 ELSE 0 END) AS cancelled_bookings,
              COALESCE(AVG(CASE WHEN booking_status IN ('confirmed', 'completed') THEN total_amount END), 0) AS avg_order_value
         FROM bookings`
    );

    const [monthly] = await pool.execute(
      `SELECT DATE_FORMAT(COALESCE(p.verified_at, p.created_at), '%Y-%m') AS month,
              COALESCE(SUM(p.amount), 0) AS revenue,
              COUNT(*) AS transactions
         FROM payments p
        WHERE p.status = 'succeeded'
        GROUP BY month
        ORDER BY month DESC
        LIMIT 12`
    );

    const [byTour] = await pool.execute(
      `SELECT t.id AS tour_id,
              t.title AS tour_title,
              COUNT(DISTINCT b.id) AS bookings_count,
              COALESCE(SUM(bi.quantity), 0) AS travelers_count,
              COALESCE(SUM(bi.line_total), 0) AS booked_value
         FROM tours t
         LEFT JOIN booking_items bi ON bi.tour_id = t.id
         LEFT JOIN bookings b ON b.id = bi.booking_id AND b.booking_status NOT IN ('cancelled', 'rejected', 'expired')
        WHERE t.deleted_at IS NULL
        GROUP BY t.id
        ORDER BY booked_value DESC
        LIMIT 15`
    );

    const gross = Number(grossRow.gross_revenue || 0);
    const refunds = Number(refundRow.total_refunds || 0);

    return sendSuccess(res, {
      kpis: {
        gross_revenue: gross,
        total_refunds: refunds,
        net_revenue: gross - refunds,
        payment_count: Number(grossRow.payment_count || 0),
        refund_count: Number(refundRow.refund_count || 0),
        total_bookings: Number(bookingStats.total_bookings || 0),
        confirmed_bookings: Number(bookingStats.confirmed_bookings || 0),
        cancelled_bookings: Number(bookingStats.cancelled_bookings || 0),
        avg_order_value: Number(bookingStats.avg_order_value || 0),
      },
      monthly: monthly.reverse(),
      byTour,
    });
  })
);

    if (!booking_id) {
      throw AppError.badRequest("booking_id is required.", "VALIDATION_ERROR");
    }

    const [[booking]] = await pool.execute(
      `SELECT id, booking_reference, customer_name, customer_email, currency_code,
              subtotal, discount_amount, tax_amount, total_amount, payment_status
         FROM bookings
        WHERE id = ?
        LIMIT 1`,
      [Number(booking_id)]
    );
    if (!booking) throw AppError.notFound("Booking not found.");

    const year = new Date().getFullYear();
    const rand = String(Math.floor(1000 + Math.random() * 9000));
    const invoiceNumber = `INV-${year}-${booking.id}-${rand}`;
    const resolvedStatus = booking.payment_status === "paid" ? "paid" : status;

    const [result] = await pool.execute(
      `INSERT INTO invoices (
         invoice_number, booking_id, customer_name, customer_email,
         currency_code, subtotal, discount_amount, tax_amount, total_amount,
         status, due_at, notes, created_by
       ) VALUES (?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?, ?)`,
      [
        invoiceNumber,
        booking.id,
        booking.customer_name,
        booking.customer_email,
        booking.currency_code || "BDT",
        booking.subtotal,
        booking.discount_amount,
        booking.tax_amount,
        booking.total_amount,
        resolvedStatus,
        due_at || null,
        notes || null,
        req.auth.userId,
      ]
    );

    const [[created]] = await pool.execute(
      `SELECT i.*, b.booking_reference, b.booking_status, b.payment_status
         FROM invoices i
         JOIN bookings b ON b.id = i.booking_id

    const [rows] = await pool.execute(

              attempt_count, last_error, received_at, last_attempt_at, processed_at,
              payload_json, result_json
         FROM payment_webhook_events
        ${sqlWhere}
        ORDER BY received_at DESC, id DESC
        LIMIT 100`,
      params
    );
    return sendSuccess(res, rows);
  })
);

const express = require('express');
const { z } = require('zod');
const asyncHandler = require('../../utils/asynchandler');
const { sendSuccess } = require('../../utils/response');
const service = require('./gallery.service');

const idSchema = z.coerce.number().int().positive();
const pagination = z.object({ page: z.coerce.number().int().min(1).default(1), limit: z.coerce.number().int().min(1).max(100).default(20) });
const itemSchema = z.object({
  tour_id: z.number().int().positive().nullable().optional(),
  title: z.string().trim().max(200).nullable().optional(),
  media_url: z.string().trim().url().max(1024),
  alt_text: z.string().trim().max(255).nullable().optional(),
  sort_order: z.number().int().min(-2147483648).max(2147483647).optional(),
  status: z.enum(['draft','published','archived']).optional()
});
const patchSchema = itemSchema.partial();
const statuses = ['draft','published','archived'];

// Inject the application's real auth and RBAC middleware. No fake auth is provided.
function createGalleryRouter({ authenticate, authorize }) {
  if (typeof authenticate !== 'function' || typeof authorize !== 'function') {
    throw new Error('Gallery router requires real requireAuth and requirePermission middleware');
  }
  const router = express.Router();

  // Admin endpoints. Mount this router at /gallery; protect /admin separately in the route index below.
  const admin = express.Router();
  admin.use(authenticate, authorize('content.manage'));
  admin.get('/', asyncHandler(async (req,res) => {
    const q = pagination.parse(req.query);
    const status = req.query.status === undefined ? undefined : z.enum(statuses).parse(req.query.status);
    const tourId = req.query.tour_id === undefined ? undefined : idSchema.parse(req.query.tour_id);
    return sendSuccess(res, await service.listAdmin({ ...q, status, tourId }));
  }));
  admin.post('/', asyncHandler(async (req,res) => {
    const data = itemSchema.parse(req.body);
    const item = await service.create(data, req.user?.id);
    return sendSuccess(res, item, { status:201 });
  }));
  admin.patch('/:id', asyncHandler(async (req,res) => {
    const id = idSchema.parse(req.params.id); const data = patchSchema.parse(req.body);
    const item = await service.update(id, data);
    if (!item) return res.status(404).json({ success:false, error:{ code:'GALLERY_ITEM_NOT_FOUND', message:'Gallery item not found' } });
    return sendSuccess(res, item);
  }));
  admin.delete('/:id', asyncHandler(async (req,res) => {
    const id = idSchema.parse(req.params.id);
    if (!(await service.remove(id))) return res.status(404).json({ success:false, error:{ code:'GALLERY_ITEM_NOT_FOUND', message:'Gallery item not found' } });
    return sendSuccess(res, { id, deleted:true });
  }));
  router.use('/admin', admin);

  // Public gallery: published items only.
  router.get('/', asyncHandler(async (req, res) => {
    const q = pagination.parse(req.query);
    const tourId = req.query.tour_id === undefined ? undefined : idSchema.parse(req.query.tour_id);
    return sendSuccess(res, await service.listPublic({ ...q, tourId }));
  }));
  router.get('/:id', asyncHandler(async (req, res) => {
    const id = idSchema.parse(req.params.id); const item = await service.getById(id);
    if (!item || item.status !== 'published') return res.status(404).json({ success:false, error:{ code:'GALLERY_ITEM_NOT_FOUND', message:'Gallery item not found' } });
    return sendSuccess(res, item);
  }));

  return router;
}
module.exports = { createGalleryRouter };

const { pool } = require('../../config/db');

const SELECT = `SELECT id, tour_id, title, media_url, alt_text, sort_order, status, created_by, created_at FROM gallery_items`;

async function listPublic({ page, limit, tourId }) {
  const where = ['status = ?'];
  const params = ['published'];
  if (tourId) { where.push('tour_id = ?'); params.push(tourId); }
  const offset = (page - 1) * limit;
  const [rows] = await pool.execute(
    `${SELECT} WHERE ${where.join(' AND ')} ORDER BY sort_order ASC, id DESC LIMIT ? OFFSET ?`,
    [...params, limit, offset]
  );
  const [[count]] = await pool.execute(`SELECT COUNT(*) AS total FROM gallery_items WHERE ${where.join(' AND ')}`, params);
  return { items: rows, pagination: { page, limit, total: Number(count.total), pages: Math.ceil(Number(count.total) / limit) } };
}

async function listAdmin({ page, limit, status, tourId }) {
  const where = ['1=1']; const params = [];
  if (status) { where.push('status = ?'); params.push(status); }
  if (tourId) { where.push('tour_id = ?'); params.push(tourId); }
  const offset = (page - 1) * limit;
  const [rows] = await pool.execute(`${SELECT} WHERE ${where.join(' AND ')} ORDER BY created_at DESC, id DESC LIMIT ? OFFSET ?`, [...params, limit, offset]);
  const [[count]] = await pool.execute(`SELECT COUNT(*) AS total FROM gallery_items WHERE ${where.join(' AND ')}`, params);
  return { items: rows, pagination: { page, limit, total: Number(count.total), pages: Math.ceil(Number(count.total) / limit) } };
}

async function create(data, userId) {
  const [result] = await pool.execute(
    `INSERT INTO gallery_items (tour_id,title,media_url,alt_text,sort_order,status,created_by) VALUES (?,?,?,?,?,?,?)`,
    [data.tour_id ?? null, data.title ?? null, data.media_url, data.alt_text ?? null, data.sort_order ?? 0, data.status ?? 'draft', userId ?? null]
  );
  return getById(result.insertId);
}
async function getById(id) {
  const [rows] = await pool.execute(`${SELECT} WHERE id = ? LIMIT 1`, [id]);
  return rows[0] || null;
}
async function update(id, data) {
  const fields = ['tour_id','title','media_url','alt_text','sort_order','status'];
  const sets = []; const values = [];
  for (const field of fields) if (Object.prototype.hasOwnProperty.call(data, field)) { sets.push(`${field} = ?`); values.push(data[field]); }
  if (!sets.length) return getById(id);
  await pool.execute(`UPDATE gallery_items SET ${sets.join(', ')} WHERE id = ?`, [...values, id]);
  return getById(id);
}
async function remove(id) { const [result] = await pool.execute('DELETE FROM gallery_items WHERE id = ?', [id]); return result.affectedRows > 0; }
module.exports = { listPublic, listAdmin, create, getById, update, remove };

const env = require("../../config/env");
const logger = require("../../utils/logger");
const { enqueue } = require("./notification.queue");

const baseUrl = () => env.FRONTEND_URL.split(",")[0].trim().replace(/\/+$/, "");
const fmtMoney = (amount, cur) =>
  `${cur} ${Number(amount).toLocaleString("en-US", { minimumFractionDigits: 2, maximumFractionDigits: 2 })}`;
const fmtDate = (d) =>
  d ? new Intl.DateTimeFormat("en-GB", { dateStyle: "medium", timeZone: "Asia/Dhaka" }).format(new Date(d)) : "";

/**
 * Queue one email about a booking. Call it with the SAME connection as the surrounding
 * transaction. It never throws: a failed email must not roll back a booking or payment.
 * The idempotency key makes retries and duplicate gateway callbacks harmless.
 */
async function queueBookingEmail(conn, bookingId, templateKey, extra = {}, keySuffix = "") {
  try {
    const [[b]] = await conn.execute(
      `SELECT id, user_id, booking_reference, customer_name, customer_email, currency_code, total_amount
         FROM bookings WHERE id = ?`, [bookingId]);
    if (!b) return;
    const [[it]] = await conn.execute(
      `SELECT bi.quantity, bi.tour_snapshot, td.starts_at
         FROM booking_items bi LEFT JOIN tour_departures td ON td.id = bi.departure_id
        WHERE bi.booking_id = ? ORDER BY bi.id LIMIT 1`, [bookingId]);

    let title = "";
    try {
      const snap = typeof it?.tour_snapshot === "string" ? JSON.parse(it.tour_snapshot) : it?.tour_snapshot;
      title = snap?.title || "";
    } catch { /* snapshot is informational only */ }

    const extras = { ...extra };
    for (const k of ["amount", "refund_amount"]) {
      if (extras[k] != null) extras[k] = fmtMoney(extras[k], b.currency_code);
    }
    const member = Boolean(b.user_id);
    const url = `${baseUrl()}/bookings/${b.id}`;
    const viewLine = member
      ? `View your booking: ${url}`
      : `Questions? Contact us: ${baseUrl()}/contact`;
    const nextStep = member
      ? `Please complete your payment within ${extras.hold_minutes || 15} minutes to keep your seats: ${url}`
      : "Our team will review your booking and contact you by email or phone to arrange payment.";
    const payload = {
      name: b.customer_name,
      booking_reference: b.booking_reference,
      tour_title: title,
      departure_date: fmtDate(it?.starts_at),
      travellers: String(it?.quantity ?? ""),
      total: fmtMoney(b.total_amount, b.currency_code),
      booking_url: url,
      view_line: viewLine,
      next_step: nextStep,
      ...extras,
    };
    await enqueue({
      userId: b.user_id,
      channel: "email",
      recipient: b.customer_email,
      templateKey,
      payload,
      idempotencyKey: `${templateKey}:${b.id}${keySuffix}`,
      conn,
    });
  } catch (error) {
    logger.error("Could not queue booking email", { templateKey, bookingId, error: error.message });
  }
}

module.exports = { queueBookingEmail };

const { pool, withTransaction } = require("../../config/db");
const { validateChannel, validateRecipient, validateKey } = require("./notification.validation");

async function enqueue({ userId = null, channel, recipient, templateKey, payload = {}, idempotencyKey = null, conn = null }) {
  validateChannel(channel);
  validateRecipient(channel, recipient);
  validateKey(templateKey);
  if (!payload || typeof payload !== "object" || Array.isArray(payload)) {
    throw Object.assign(new Error("payload must be an object"), { statusCode: 400 });
  }
  if (idempotencyKey !== null && (typeof idempotencyKey !== "string" || idempotencyKey.length > 190)) {
    throw Object.assign(new Error("Invalid idempotency key"), { statusCode: 400 });
  }
  const db = conn || pool;
  const [result] = await db.execute(
    `INSERT INTO notification_jobs
      (user_id, channel, recipient, template_key, payload_json, idempotency_key)
     VALUES (?, ?, ?, ?, ?, ?)
     ON DUPLICATE KEY UPDATE id = LAST_INSERT_ID(id)`,
    [userId, channel, recipient, templateKey, JSON.stringify(payload), idempotencyKey]);
  return { jobId: Number(result.insertId), queued: true };
}
async function enqueueInTransaction(args, conn) {
  return enqueue({ ...args, conn });
}
module.exports = { enqueue, enqueueInTransaction };

async function enqueueForUser({ userId, templateKey, payload, channels = ["email", "sms"], email, phone, conn = null }) {
  const [rows] = await (conn || pool).execute(
    "SELECT email, phone FROM users WHERE id = ? LIMIT 1", [userId]);
  const user = rows[0];
  if (!user) throw Object.assign(new Error("User not found"), { statusCode: 404 });
  const jobs = [];
  for (const channel of channels) {
    const recipient = channel === "email" ? (email || user.email) : (phone || user.phone);
    if (!recipient) continue;
    jobs.push(await enqueue({ userId, channel, recipient, templateKey, payload, conn }));
  }
  return jobs;
}

function render(template, payload) {
  return String(template).replace(/\{\{([a-zA-Z0-9_]+)\}\}/g, (match, key) => {
    const value = payload && Object.prototype.hasOwnProperty.call(payload, key) ? payload[key] : "";
    return value == null ? "" : String(value);
  });
}
function renderTemplate(row, payload) {
  return {
    subject: row.subject_template ? render(row.subject_template, payload) : undefined,
    body: render(row.body_template, payload)
  };
}

const PHONE = /^\+[1-9]\d{7,14}$/; // E.164 format
const KEY = /^[a-z][a-z0-9_]{1,99}$/;
function validateChannel(channel) {
  if (!["email", "sms"].includes(channel)) throw Object.assign(new Error("channel must be email or sms"), { statusCode: 400 });
  return channel;
}
function validateRecipient(channel, recipient) {
  if (typeof recipient !== "string" || recipient.length > 190) throw Object.assign(new Error("Invalid recipient"), { statusCode: 400 });
  const value = recipient.trim();
  if (channel === "email" && !EMAIL.test(value)) throw Object.assign(new Error("A valid email recipient is required"), { statusCode: 400 });
  if (channel === "sms" && !PHONE.test(value)) throw Object.assign(new Error("SMS recipient must be in E.164 format, e.g. +8801XXXXXXXXX"), { statusCode: 400 });
  return value;
}
function validateKey(key) {
  if (typeof key !== "string" || !KEY.test(key)) throw Object.assign(new Error("Invalid notification template key"), { statusCode: 400 });
  return key;
}
module.exports = { validateChannel, validateRecipient, validateKey };

const { pool, withTransaction } = require("../../config/db");
const { renderTemplate } = require("./notification.templates");
const emailProvider = require("./providers/email.provider");
const smsProvider = require("./providers/sms.provider");

async function claimOne() {
  return withTransaction(async (conn) => {
    await conn.execute(
      `UPDATE notification_jobs SET status = 'queued', locked_at = NULL
        WHERE status = 'processing' AND locked_at < DATE_SUB(NOW(), INTERVAL 10 MINUTE)`
    );
    const [rows] = await conn.execute(
      `SELECT * FROM notification_jobs
        WHERE status = 'queued' AND available_at <= NOW() AND attempts < max_attempts
        ORDER BY available_at ASC, id ASC LIMIT 1 FOR UPDATE`
    );
    if (!rows.length) return null;
    const job = rows[0];
    await conn.execute(
      `UPDATE notification_jobs SET status = 'processing', locked_at = NOW(), attempts = attempts + 1
        WHERE id = ?`,
      [job.id]
    );
    return { ...job, attempts: Number(job.attempts) + 1 };
  });
}

async function processOne() {
  const job = await claimOne();
  if (!job) return false;
  try {
    const [[template]] = await pool.execute(
      `SELECT * FROM notification_templates
        WHERE template_key = ? AND channel = ? AND is_active = 1`,
      [job.template_key, job.channel]
    );
    if (!template) throw new Error(`Active template not found: ${job.template_key}/${job.channel}`);

    let payload;
    try {
      payload = JSON.parse(job.payload_json);
    } catch (_) {
      throw new Error("Notification payload is not valid JSON");
    }

    const rendered = renderTemplate(template, payload);
    const isCampaign = job.template_key === "marketing_campaign" && payload.html;
    const plainText = isCampaign
      ? String(rendered.body).replace(/<[^>]+>/g, " ").replace(/\s+/g, " ").trim()
      : rendered.body;

    const result =
      job.channel === "email"
        ? await emailProvider.send({
            to: job.recipient,
            subject: rendered.subject || "",
            text: plainText,
            ...(isCampaign ? { html: rendered.body } : {}),
          })
        : await smsProvider.send({ to: job.recipient, text: rendered.body });

    await withTransaction(async (conn) => {
      await conn.execute(
        `UPDATE notification_jobs SET status = 'sent', sent_at = NOW(), locked_at = NULL, last_error = NULL
          WHERE id = ?`,
        [job.id]
      );
      await conn.execute(
        `INSERT INTO notification_attempts (job_id, attempt_number, outcome, provider_message_id)
         VALUES (?, ?, 'sent', ?)
         ON DUPLICATE KEY UPDATE
           outcome = 'sent',
           provider_message_id = VALUES(provider_message_id),
           error_message = NULL`,
        [job.id, job.attempts, result && result.messageId ? String(result.messageId).slice(0, 190) : null]
      );
    });
  } catch (error) {
    const message = String((error && error.message) || "Delivery failed").slice(0, 1000);
    const retry = job.attempts < Number(job.max_attempts);
    const delaySeconds = Math.min(3600, 30 * 2 ** Math.max(0, job.attempts - 1));
    await withTransaction(async (conn) => {
      await conn.execute(
        `UPDATE notification_jobs SET status = ?, locked_at = NULL, last_error = ?,
          available_at = CASE WHEN ? = 1 THEN DATE_ADD(NOW(), INTERVAL ? SECOND) ELSE available_at END
          WHERE id = ?`,
        [retry ? "queued" : "failed", message, retry ? 1 : 0, delaySeconds, job.id]
      );
      await conn.execute(
        `INSERT INTO notification_attempts (job_id, attempt_number, outcome, error_message)
         VALUES (?, ?, 'failed', ?)
         ON DUPLICATE KEY UPDATE
           outcome = 'failed',
           error_message = VALUES(error_message)`,
        [job.id, job.attempts, message]
      );
    });
  }
  return true;
}

async function runWorker({ pollMs = 3000, batchSize = 10 } = {}) {
  let stopped = false;
  const stop = () => {
    stopped = true;
  };
  async function loop() {
    while (!stopped) {
      let worked = false;
      for (let i = 0; i < batchSize && !stopped; i++) {
        const didWork = await processOne();
        if (!didWork) break;
        worked = true;
      }
      if (!worked) await new Promise((resolve) => setTimeout(resolve, pollMs));
    }
  }
  const promise = loop();
  return { stop, promise };
}

module.exports = { processOne, runWorker };

const crypto = require('crypto');
const { pool, withTransaction } = require('../../config/db');
const sslcommerz = require('../sslcommerz/sslcommerz.service');

function problem(statusCode, message) { const e = new Error(message); e.statusCode=statusCode; return e; }
function canonical(payload) {
  return Object.keys(payload || {}).sort().reduce((out,k) => {
    const v=payload[k];
    if (v !== undefined && v !== null && typeof v !== 'object') out[k]=String(v);
    return out;
  }, {});
}
function eventIdentity(payload) {
  const normalized=canonical(payload);
  // Prefer gateway's validation ID; include payload digest to distinguish malformed/replayed variants.
  const source = normalized.val_id ? `val_id:${normalized.val_id}` :
    normalized.tran_id ? `tran_id:${normalized.tran_id}:${normalized.status || ''}` :
    JSON.stringify(normalized);
  return crypto.createHash('sha256').update(source).digest('hex');
}
async function receive(payload) {
  if (!payload || typeof payload !== 'object' || Array.isArray(payload)) throw problem(400,'Invalid IPN payload.');
  const valId=String(payload.val_id || '').trim();
  if (!valId) throw problem(400,'SSLCommerz IPN is missing val_id.');
  if (valId.length>100) throw problem(400,'Invalid SSLCommerz validation ID.');
  const eventKey=eventIdentity(payload);
  const raw=JSON.stringify(payload);
  const [insert] = await pool.execute(
    `INSERT IGNORE INTO payment_webhook_events
      (provider,event_key,validation_id,payload_json,processing_status,attempt_count,received_at)
     VALUES ('sslcommerz',?,?,?,'received',0,UTC_TIMESTAMP())`,
    [eventKey,valId,raw]
  );
  const [rows]=await pool.execute(
    'SELECT id,processing_status,attempt_count FROM payment_webhook_events WHERE provider = ? AND event_key = ?',
    ['sslcommerz',eventKey]
  );
  const event=rows[0];
  if (!event) throw problem(500,'Could not persist webhook event.');
  if (event.processing_status==='processed') return { accepted:true, duplicate:true, event_id:event.id };
  // Synchronous first attempt keeps SSLCommerz response quick; durable row remains for worker retries.
  await processEvent(event.id);
  const [[latest]]=await pool.execute('SELECT processing_status FROM payment_webhook_events WHERE id=?',[event.id]);
  return { accepted:true, duplicate:insert.affectedRows===0, event_id:event.id, processing_status:latest.processing_status };
}
async function processEvent(eventId) {
  const claimed=await withTransaction(async conn=>{
    const [rows]=await conn.execute(
      `SELECT * FROM payment_webhook_events WHERE id=? FOR UPDATE`,[eventId]
    );
    if(!rows.length) return null;
    const ev=rows[0];
    if(ev.processing_status==='processed' || ev.processing_status==='processing') return null;
    await conn.execute(
      `UPDATE payment_webhook_events SET processing_status='processing',
        attempt_count=attempt_count+1,last_attempt_at=UTC_TIMESTAMP(),last_error=NULL WHERE id=?`,
      [eventId]
    );
    return {
      payload_json: ev.payload_json,
      attempt_count: Number(ev.attempt_count) + 1
    };
  });
  if(!claimed) return { skipped:true };
  try {
    const payload=typeof claimed.payload_json==='string'
      ? JSON.parse(claimed.payload_json)
      : claimed.payload_json;
    const result=await sslcommerz.validateAndApply(payload);
    await pool.execute(
      `UPDATE payment_webhook_events SET processing_status='processed',processed_at=UTC_TIMESTAMP(),
        result_json=?,last_error=NULL,next_retry_at=NULL WHERE id=?`,
      [JSON.stringify(result),eventId]
    );
    return result;
  } catch(error) {
    const retryDelay = Math.min(3600, 30 * (2 ** Math.max(0, claimed.attempt_count - 1)));
    await pool.execute(
      `UPDATE payment_webhook_events SET processing_status='failed',last_error=?,
        next_retry_at=DATE_ADD(UTC_TIMESTAMP(), INTERVAL ? SECOND) WHERE id=?`,
      [String(error.message||'Webhook processing failed').slice(0,1000),retryDelay,eventId]
    );
    throw error;
  }
}
async function retryFailed(limit=25) {
  const safeLimit=Math.max(1,Math.min(Number(limit)||25,100));
  const [rows]=await pool.execute(
    `SELECT id FROM payment_webhook_events
      WHERE processing_status IN ('received','failed')
        AND attempt_count < 12
        AND (next_retry_at IS NULL OR next_retry_at <= UTC_TIMESTAMP())
      ORDER BY received_at ASC LIMIT ${safeLimit}`

      results.push({id:row.id,ok:true});
    } catch(error) { results.push({id:row.id,ok:false,error:error.message}); }
  }

}

const ALLOWED_STATUSES = new Set(["initiated", "pending", "succeeded", "failed", "cancelled"]);

function httpError(status, message) {
  const error = new Error(message);
  error.statusCode = status;
  return error;
}
function positiveId(value) {
  if (!/^[1-9]\d*$/.test(String(value))) throw httpError(400, "Invalid payment id.");
  return String(value);
}
function getPool() {
  return db.pool || db;
}
async function query(sql, params = [], conn = null) {
  const target = conn || getPool();
  if (typeof target.execute === "function") return target.execute(sql, params);
  if (typeof target.query === "function") return target.query(sql, params);
  throw new Error("Database module must export a mysql2 pool or connection.");
}
async function rows(sql, params = [], conn = null) {
  const result = await query(sql, params, conn);
  return Array.isArray(result) && Array.isArray(result[0]) ? result[0] : result;
}
async function getConnection() {
  const pool = getPool();
  return typeof pool.getConnection === "function" ? pool.getConnection() : null;

async function list(queryParams) {
  const page = Math.max(1, Math.min(100000, Number.parseInt(queryParams.page, 10) || 1));
  const limit = Math.max(1, Math.min(100, Number.parseInt(queryParams.limit, 10) || 20));
  const where = [];
  const params = [];
  if (queryParams.status) {
    if (!ALLOWED_STATUSES.has(queryParams.status)) throw httpError(400, "Invalid payment status.");
    where.push("p.status = ?");
    params.push(queryParams.status);
  }
  if (queryParams.booking_id) {
    if (!/^[1-9]\d*$/.test(String(queryParams.booking_id))) throw httpError(400, "Invalid booking_id.");
    where.push("p.booking_id = ?");
    params.push(queryParams.booking_id);
  }
  if (queryParams.provider) {
    where.push("p.provider = ?");
    params.push(String(queryParams.provider).slice(0, 50));
  }
  if (queryParams.from) {
    if (Number.isNaN(Date.parse(queryParams.from))) throw httpError(400, "Invalid from date.");
    where.push("p.created_at >= ?");
    params.push(queryParams.from);
  }
  if (queryParams.to) {
    if (Number.isNaN(Date.parse(queryParams.to))) throw httpError(400, "Invalid to date.");
    where.push("p.created_at < DATE_ADD(?, INTERVAL 1 DAY)");
    params.push(queryParams.to);
  }
  const clause = where.length ? `WHERE ${where.join(" AND ")}` : "";
  const countRows = await rows(`SELECT COUNT(*) AS total FROM payments p ${clause}`, params);
  const data = await rows(
    `SELECT p.id, p.booking_id, b.booking_reference, p.provider, p.provider_payment_id,
            p.amount, p.currency_code, p.status, p.verified_at, p.created_at, p.updated_at
       FROM payments p
       LEFT JOIN bookings b ON b.id = p.booking_id
       ${clause}
      ORDER BY p.created_at DESC, p.id DESC LIMIT ? OFFSET ?`,
    [...params, limit, (page - 1) * limit]
  );
  return { data, pagination: { page, limit, total: Number(countRows[0]?.total || 0), pages: Math.ceil(Number(countRows[0]?.total || 0) / limit) } };
}
async function getById(id, conn = null) {
  id = positiveId(id);
  const found = await rows(
    `SELECT p.id, p.booking_id, b.booking_reference, b.booking_status, b.payment_status AS booking_payment_status,
            p.provider, p.provider_payment_id, p.idempotency_key, p.amount, p.currency_code,
            p.status, p.verified_at, p.created_at, p.updated_at
       FROM payments p LEFT JOIN bookings b ON b.id = p.booking_id WHERE p.id = ? LIMIT 1`,
    [id], conn
  );
  if (!found.length) throw httpError(404, "Payment not found.");
  return found[0];
}
async function mutate(id, actorId, note, targetStatus) {
  id = positiveId(id);
  const conn = await getConnection();
  const run = async (c) => {
    const found = await rows("SELECT id, booking_id, status FROM payments WHERE id = ? FOR UPDATE", [id], c);
    if (!found.length) throw httpError(404, "Payment not found.");
    const payment = found[0];
    if (payment.status === targetStatus) return getById(id, c);
    if (payment.status === "succeeded" || payment.status === "cancelled") {
      throw httpError(409, `Cannot change a ${payment.status} payment through this endpoint.`);
    }
    if (targetStatus === "succeeded") {
      await query("UPDATE payments SET status = 'succeeded', verified_at = CURRENT_TIMESTAMP WHERE id = ?", [id], c);
      await query("UPDATE bookings SET payment_status = 'paid' WHERE id = ?", [payment.booking_id], c);
    } else {
      await query("UPDATE payments SET status = 'failed' WHERE id = ?", [id], c);
      const remaining = await rows(
        "SELECT COUNT(*) AS count FROM payments WHERE booking_id = ? AND status = 'succeeded' AND id <> ?",
        [payment.booking_id, id], c
      );
      if (Number(remaining[0]?.count || 0) === 0) {
        await query("UPDATE bookings SET payment_status = 'failed' WHERE id = ?", [payment.booking_id], c);
      }
    }
    // Keep a record of the administrative action without storing payment-provider payloads.
    try {
      await query(
        `INSERT INTO audit_logs
           (actor_user_id, action, entity_type, entity_id, before_data, after_data)
         VALUES (?, ?, 'payment', ?, ?, ?)`,
        [actorId, targetStatus === "succeeded" ? "payment.admin_verify" : "payment.admin_mark_failed", id,
         JSON.stringify({ status: payment.status }),
         JSON.stringify({
           bookingId: payment.booking_id,
           status: targetStatus,
           note: note ? String(note).slice(0, 500) : null,
         })], c
      );
    } catch (e) {
      // Some existing schemas use a different audit_logs shape; payment state remains transactional
      // and the caller still receives an error if the database transaction itself fails.
      if (e.code !== "ER_BAD_FIELD_ERROR" && e.code !== "ER_NO_SUCH_TABLE") throw e;
    }
    return getById(id, c);
  };
  if (!conn) return run(null);
  try {
    await conn.beginTransaction();
    const result = await run(conn);
    await conn.commit();
    return result;
  } catch (error) {
    await conn.rollback();
    throw error;
  } finally {
    conn.release();
  }
}

    `SELECT COUNT(*) AS total FROM roles r ${where}`,

      FROM roles r

    const role = await findRole(id, connection);
    if (!role) return null;
    const before = { name: role.name, description: role.description, permissions: role.permissions };
    const nextName = name === undefined ? role.name : name;
    const nextDescription = description === undefined ? role.description : description;
    await connection.execute(
      "UPDATE roles SET name = ?, description = ? WHERE id = ?",
      [nextName, nextDescription, id],
    );
    const updated = await findRole(id, connection);
    await writeAudit(connection, {
      actorId,
      action: "rbac.role.update",
      entityId: id,
      request,
      before,
      after: {
        name: updated.name,
        description: updated.description,
        permissions: updated.permissions,
      },
    });
    return updated;
  });
}

async function replaceRolePermissions({ id, permissionIds, permissionKeys, actorId, request }) {
  return withTransaction(async (connection) => {
    const [[lockedRole]] = await connection.execute(
      "SELECT id, role_key FROM roles WHERE id = ? FOR UPDATE",
      [id],
    );
    if (!lockedRole) return null;
    const role = await findRole(id, connection);
    await connection.execute("DELETE FROM role_permissions WHERE role_id = ?", [id]);
    for (const permissionId of permissionIds) {
      await connection.execute(
        "INSERT INTO role_permissions (role_id, permission_id) VALUES (?, ?)",
        [id, permissionId],
      );
    }
    await writeAudit(connection, {
      actorId,
      action: "rbac.role.permissions.replace",
      entityId: id,
      request,
      before: { permissions: role.permissions },
      after: { permissions: permissionKeys },
    });
    return findRole(id, connection);
  });
}

async function assignRoleToUser({ roleId, userId, actorId, request }) {
  return withTransaction(async (connection) => {
    const [[user]] = await connection.execute(
      "SELECT id FROM users WHERE id = ? AND account_status <> 'deleted' FOR UPDATE",
      [userId],
    );
    if (!user) return null;
    const [[role]] = await connection.execute(
      "SELECT id, role_key FROM roles WHERE id = ? FOR UPDATE",
      [roleId],
    );
    if (!role) return false;
    const [[existing]] = await connection.execute(
      "SELECT user_id FROM user_roles WHERE user_id = ? AND role_id = ? FOR UPDATE",
      [userId, roleId],
    );
    if (existing) return { userId, roleId, alreadyAssigned: true };
    await connection.execute(
      "INSERT INTO user_roles (user_id, role_id, assigned_by) VALUES (?, ?, ?)",
      [userId, roleId, actorId],
    );
    await writeAudit(connection, {
      actorId,
      action: "rbac.role.user.assign",
      entityId: userId,
      request,
      before: null,
      after: { roleId, roleKey: role.role_key, userId },
    });
    return { userId, roleId, alreadyAssigned: false };
  });
}

async function removeRoleFromUser({ roleId, userId, actorId, request }) {

      [roleId],
    );
    if (!role) return false;
    const [deleted] = await connection.execute(
      "DELETE FROM user_roles WHERE user_id = ? AND role_id = ?",

    );

    await writeAudit(connection, {
      actorId,
      action: "rbac.role.user.remove",
      entityId: userId,
      request,
      before: { roleId, roleKey: role.role_key, userId },
      after: null,
    });
    return { userId, roleId, removed: true };
  });
}

async function writeAudit(connection, { actorId, action, entityId, request, before = null, after }) {
  await connection.execute(
    `INSERT INTO audit_logs
       (actor_user_id, action, entity_type, entity_id, request_id, ip_address, user_agent, before_data, after_data)
     VALUES (?, ?, 'rbac', ?, ?, INET6_ATON(?), ?, ?, ?)`,
    [
      actorId,
      action,
      entityId,
      typeof request?.id === "string" ? request.id.slice(0, 100) : null,
      typeof request?.ip === "string" ? request.ip.slice(0, 45) : null,
      typeof request?.get === "function" ? (request.get("user-agent") || "").slice(0, 500) || null : null,
      before === null ? null : JSON.stringify(before),
      JSON.stringify(after),
    ],
  );
}

async function resolvePermissionIds(keys) {

  const [rows] = await pool.execute(
    `SELECT id, permission_key FROM permissions WHERE permission_key IN (${placeholders})`,
    keys,
  );
  return rows;
}

function assertAllowedPermissionGrants(keys, actorAccess) {
  const outsideScope = keys.filter((key) => !actorAccess.permissions.has(key));
  if (outsideScope.length) {
    throw AppError.forbidden(
      "You cannot grant permissions that you do not hold.",
      "PERMISSION_GRANT_OUT_OF_SCOPE",
    );
  }
  if (keys.includes("roles.manage") && !actorAccess.isSuperAdmin) {
    throw AppError.forbidden(
      "Only a Super Admin can delegate role-management access.",
      "SUPER_ADMIN_REQUIRED",
    );
  }
}

async function getPermissionIds(keys, actorAccess) {
  assertAllowedPermissionGrants(keys, actorAccess);
  const rows = await repository.resolvePermissionIds(keys);
  if (rows.length !== keys.length) {
    const knownKeys = new Set(rows.map((row) => row.permission_key));
    const unknown = keys.filter((key) => !knownKeys.has(key));
    throw AppError.badRequest(`Unknown permission key: ${unknown.join(", ")}.`, undefined, "UNKNOWN_PERMISSION");
  }
  return new Map(rows.map((row) => [row.permission_key, row.id]));
}

const { pool, withTransaction } = require('../../config/db');
const { queueBookingEmail } = require('../notifications/booking-events');

function problem(statusCode, message) {
  const error = new Error(message);
  error.statusCode = statusCode;
  return error;
}

function money(value) {
  const n = Number(value);
  if (!Number.isFinite(n)) return null;
  return n.toFixed(2);
}

async function refreshPaymentAndBooking(conn, paymentId, bookingId) {
  const [[payment]] = await conn.execute(
    'SELECT amount, status FROM payments WHERE id = ? FOR UPDATE', [paymentId]
  );
  const [[totals]] = await conn.execute(
    `SELECT COALESCE(SUM(CASE WHEN status IN ('requested','pending','succeeded') THEN amount ELSE 0 END), 0) AS committed,
            COALESCE(SUM(CASE WHEN status = 'succeeded' THEN amount ELSE 0 END), 0) AS succeeded
       FROM refunds WHERE payment_id = ?`, [paymentId]
  );
  const paid = Number(payment.amount);
  const committed = Number(totals.committed);
  const refunded = Number(totals.succeeded);
  if (committed - paid > 0.0001) throw problem(409, 'Committed refunds exceed the original payment amount.');

  if (payment.status === 'succeeded') {
    if (refunded >= paid - 0.005) {
      await conn.execute("UPDATE bookings SET payment_status = 'refunded' WHERE id = ?", [bookingId]);
    } else if (refunded > 0) {
      await conn.execute("UPDATE bookings SET payment_status = 'partially_refunded' WHERE id = ?", [bookingId]);
    } else {
      await conn.execute("UPDATE bookings SET payment_status = 'paid' WHERE id = ?", [bookingId]);
    }
  }
  return { original_amount: money(paid), committed_refund_amount: money(committed), refunded_amount: money(refunded), refundable_amount: money(Math.max(0, paid - committed)) };
}

async function createRefund(adminId, paymentId, amount, reason) {
  const requested = money(amount);
  if (!requested || Number(requested) <= 0) throw problem(400, 'Refund amount must be greater than zero.');
  if (reason !== undefined && (typeof reason !== 'string' || reason.length > 500)) throw problem(400, 'Reason must be 500 characters or fewer.');

  return withTransaction(async (conn) => {
    const [rows] = await conn.execute(
      `SELECT p.id, p.booking_id, p.amount, p.currency_code, p.status
         FROM payments p WHERE p.id = ? FOR UPDATE`, [paymentId]
    );
    if (!rows.length) throw problem(404, 'Payment not found.');
    const payment = rows[0];
    if (payment.status !== 'succeeded') throw problem(409, 'Only successfully verified payments can be refunded.');
    const [existing] = await conn.execute(
      `SELECT COALESCE(SUM(amount), 0) AS committed
         FROM refunds WHERE payment_id = ? AND status IN ('requested','pending','succeeded')`, [paymentId]
    );
    const remaining = Number(payment.amount) - Number(existing[0].committed);
    if (Number(requested) > remaining + 0.0001) throw problem(409, `Refund exceeds the remaining refundable amount (${money(Math.max(0, remaining))} ${payment.currency_code}).`);
    const [insert] = await conn.execute(
      `INSERT INTO refunds (payment_id, amount, reason, status, requested_by)
       VALUES (?, ?, ?, 'requested', ?)`, [paymentId, requested, reason || null, adminId]
    );
    const balance = await refreshPaymentAndBooking(conn, paymentId, payment.booking_id);
    return { id: insert.insertId, payment_id: paymentId, amount: requested, currency_code: payment.currency_code, status: 'requested', ...balance };
  });
}

async function updateRefund(adminId, refundId, status, providerRefundId) {
  const allowed = ['pending', 'succeeded', 'failed', 'cancelled'];
  if (!allowed.includes(status)) throw problem(400, `Status must be one of: ${allowed.join(', ')}.`);
  if (providerRefundId !== undefined && (typeof providerRefundId !== 'string' || providerRefundId.length > 190)) throw problem(400, 'provider_refund_id must be 190 characters or fewer.');

  return withTransaction(async (conn) => {
    const [rows] = await conn.execute(
      `SELECT r.*, p.booking_id, p.currency_code, p.amount AS payment_amount, p.status AS payment_status
         FROM refunds r JOIN payments p ON p.id = r.payment_id
        WHERE r.id = ? FOR UPDATE`, [refundId]
    );
    if (!rows.length) throw problem(404, 'Refund not found.');
    const refund = rows[0];
    if (['succeeded', 'failed', 'cancelled'].includes(refund.status)) {
      if (refund.status === status && (!providerRefundId || providerRefundId === refund.provider_refund_id)) return refund;
      throw problem(409, 'A completed refund record cannot be changed. Create a new refund if another refund is required.');
    }
    if (status === 'succeeded' && refund.payment_status !== 'succeeded') throw problem(409, 'The original payment is not in a refundable state.');

    await conn.execute(
      `UPDATE refunds SET status = ?, provider_refund_id = COALESCE(?, provider_refund_id),
              processed_by = ?, processed_at = CASE WHEN ? IN ('succeeded','failed','cancelled') THEN UTC_TIMESTAMP() ELSE processed_at END
        WHERE id = ?`, [status, providerRefundId || null, adminId, status, refundId]
    );
    const balance = await refreshPaymentAndBooking(conn, refund.payment_id, refund.booking_id);
    if (status === 'succeeded') {
      await queueBookingEmail(conn, refund.booking_id, 'refund_completed', { refund_amount: refund.amount }, `:r${refundId}`);
    }
    return { id: refundId, payment_id: refund.payment_id, amount: refund.amount, currency_code: refund.currency_code, status, provider_refund_id: providerRefundId || refund.provider_refund_id, ...balance };
  });
}

async function listRefunds({ page = 1, limit = 20, status, paymentId } = {}) {

  if (status) { where.push('r.status = ?'); params.push(status); }
  if (paymentId) { where.push('r.payment_id = ?'); params.push(paymentId); }
  const clause = where.length ? `WHERE ${where.join(' AND ')}` : '';
  const [[count]] = await pool.execute(`SELECT COUNT(*) AS total FROM refunds r ${clause}`, params);
  const [rows] = await pool.execute(
    `SELECT r.id, r.payment_id, r.amount, r.reason, r.status, r.provider_refund_id,
            r.requested_by, r.processed_by, r.processed_at, r.created_at,
            p.booking_id, p.provider, p.provider_payment_id, p.currency_code,
            b.booking_reference, b.customer_name, b.customer_email
       FROM refunds r JOIN payments p ON p.id = r.payment_id
       JOIN bookings b ON b.id = p.booking_id ${clause}
      ORDER BY r.created_at DESC LIMIT ? OFFSET ?`, [...params, l, (p - 1) * l]
  );

async function gatewayGet(params) {

/**
 * A refund is reserved in the local ledger before the network call. Once an
 * initiation request may have reached SSLCommerz, it is NEVER blindly resent.
 * Operators must query its status using the returned refund_ref_id.
 */
async function initiate(adminId, refundId) {
  const prepared = await withTransaction(async (conn) => {
    const [rows] = await conn.execute(
      `SELECT r.*, p.provider, p.provider_payment_id, p.raw_event,
              p.currency_code, p.status AS payment_status
         FROM refunds r JOIN payments p ON p.id = r.payment_id
        WHERE r.id = ? FOR UPDATE`, [refundId]
    );
    if (!rows.length) throw problem(404, 'Refund not found.');
    const r = rows[0];
    if (r.provider !== 'sslcommerz') throw problem(409, 'This refund is not associated with SSLCommerz.');
    if (r.status !== 'requested') {
      if (r.status === 'pending' && r.provider_refund_trans_id) {
        return { reused: true, id: r.id, status: r.status, refund_trans_id: r.provider_refund_trans_id,
          refund_ref_id: r.provider_refund_id };
      }
      throw problem(409, 'Only a new requested refund can be submitted to SSLCommerz.');
    }
    if (r.payment_status !== 'succeeded') throw problem(409, 'The original payment is not verified as successful.');
    let raw = {};
    try { raw = r.raw_event ? (typeof r.raw_event === 'string' ? JSON.parse(r.raw_event) : r.raw_event) : {}; } catch (_) {}
    const bankTranId = raw?.validation?.bank_tran_id || raw?.validation?.bank_tranid || raw?.bank_tran_id;
    if (!bankTranId) throw problem(409, 'The verified payment has no bank_tran_id. Reconcile the original transaction before refunding.');
    const refundTransId = `BBR${Date.now()}${crypto.randomBytes(5).toString('hex').toUpperCase()}`.slice(0, 30);
    await conn.execute(
      `UPDATE refunds SET status = 'pending', provider_refund_trans_id = ?,
              processed_by = ?, processed_at = UTC_TIMESTAMP()
        WHERE id = ? AND status = 'requested'`,
      [refundTransId, adminId, refundId]
    );
    return {
      id: r.id, payment_id: r.payment_id, amount: money(r.amount),
      reason: r.reason || 'Customer refund', bankTranId,
      refundTransId, currency_code: r.currency_code
    };
  });

  if (prepared.reused) return prepared;

  let result;
  try {
    result = await gatewayGet({
      bank_tran_id: prepared.bankTranId,
      refund_trans_id: prepared.refundTransId,
      refund_amount: prepared.amount,
      refund_remarks: String(prepared.reason).slice(0, 255),
      refe_id: String(prepared.id),
      v: '1'
    });
  } catch (error) {
    // Keep pending: timeout/network errors are ambiguous and must be reconciled,
    // not retried with a second refund transaction ID.
    await pool.execute(
      `UPDATE refunds SET provider_refund_response = ? WHERE id = ?`,
      [JSON.stringify({ transport_error: error.message, submitted_at: new Date().toISOString() }), prepared.id]
    );
    throw problem(502, 'Refund submission outcome is unknown. Do not submit again; reconcile this refund before taking further action.');
  }

  const apiConnect = String(result.APIConnect || '').toUpperCase();
  const status = String(result.status || '').toLowerCase();
  if (apiConnect !== 'DONE') {
    await pool.execute(
      `UPDATE refunds SET provider_refund_response = ? WHERE id = ?`,
      [JSON.stringify(result), prepared.id]
    );
    throw problem(502, `SSLCommerz refund API connection status: ${apiConnect || 'UNKNOWN'}. Refund remains pending for reconciliation.`);
  }

  if (status === 'failed') {
    await pool.execute(
      `UPDATE refunds SET status = 'failed', provider_refund_response = ?,
              processed_at = UTC_TIMESTAMP() WHERE id = ? AND status = 'pending'`,
      [JSON.stringify(result), prepared.id]
    );
    return { refund_id: prepared.id, status: 'failed', gateway_response: result };
  }

  await pool.execute(
    `UPDATE refunds SET provider_refund_id = ?, provider_refund_response = ?,
            last_checked_at = UTC_TIMESTAMP()
      WHERE id = ? AND status = 'pending'`,
    [result.refund_ref_id || null, JSON.stringify(result), prepared.id]
  );
  return {
    refund_id: prepared.id,
    status: 'pending',
    provider_refund_id: result.refund_ref_id || null,
    gateway_status: status || 'unknown',
    message: status === 'success' ? 'Refund initiated; final settlement must be confirmed by querying refund status.' :
      'Gateway reports the request is processing or its status is not final.'
  };
}

async function reconcile(adminId, refundId) {
  const [rows] = await pool.execute(
    `SELECT r.id, r.status, r.provider_refund_id, r.provider_refund_trans_id,
            p.id AS payment_id, p.booking_id
       FROM refunds r JOIN payments p ON p.id = r.payment_id
      WHERE r.id = ? AND p.provider = 'sslcommerz'`, [refundId]
  );
  if (!rows.length) throw problem(404, 'SSLCommerz refund not found.');
  const refund = rows[0];
  if (!refund.provider_refund_id) {
    throw problem(409, 'No SSLCommerz refund_ref_id is stored yet. Resolve the initiation outcome with the gateway/merchant support before retrying.');
  }
  const result = await gatewayGet({ refund_ref_id: refund.provider_refund_id });
  if (String(result.APIConnect || '').toUpperCase() !== 'DONE') {
    throw problem(502, `SSLCommerz refund query failed: ${result.APIConnect || 'UNKNOWN'}.`);
  }
  const gatewayStatus = String(result.status || '').toLowerCase();
  let nextStatus = null;
  if (gatewayStatus === 'refunded') nextStatus = 'succeeded';
  else if (gatewayStatus === 'cancelled') nextStatus = 'cancelled';
  else if (gatewayStatus === 'failed') nextStatus = 'failed';
  else if (gatewayStatus === 'processing') nextStatus = 'pending';

  if (!nextStatus) throw problem(502, `Unrecognized SSLCommerz refund status: ${gatewayStatus || 'empty'}.`);

  return withTransaction(async (conn) => {
    const [locked] = await conn.execute(
      `SELECT r.*, p.booking_id FROM refunds r JOIN payments p ON p.id = r.payment_id
        WHERE r.id = ? FOR UPDATE`, [refundId]
    );
    if (!locked.length) throw problem(404, 'Refund not found.');
    const current = locked[0];
    if (['succeeded', 'failed', 'cancelled'].includes(current.status) && current.status !== nextStatus) {
      throw problem(409, 'Gateway status conflicts with a terminal local refund status. Manual investigation is required.');
    }
    await conn.execute(
      `UPDATE refunds SET status = ?, provider_refund_response = ?,
              last_checked_at = UTC_TIMESTAMP(), processed_by = ?,
              processed_at = CASE WHEN ? IN ('succeeded','failed','cancelled') THEN UTC_TIMESTAMP() ELSE processed_at END
        WHERE id = ?`,
      [nextStatus, JSON.stringify(result), adminId, nextStatus, refundId]
    );
    if (nextStatus === 'succeeded') {
      await queueBookingEmail(conn, current.booking_id, 'refund_completed', { refund_amount: current.amount }, `:r${refundId}`);
    }

    const [[payment]] = await conn.execute(
      'SELECT amount, status FROM payments WHERE id = ? FOR UPDATE', [current.payment_id]
    );
    const [[sums]] = await conn.execute(
      `SELECT COALESCE(SUM(CASE WHEN status IN ('requested','pending','succeeded') THEN amount ELSE 0 END),0) AS committed,
              COALESCE(SUM(CASE WHEN status = 'succeeded' THEN amount ELSE 0 END),0) AS refunded
         FROM refunds WHERE payment_id = ?`, [current.payment_id]
    );
    if (Number(sums.committed) - Number(payment.amount) > 0.0001) {
      throw problem(409, 'Committed refunds exceed original payment amount; manual reconciliation required.');
    }
    if (payment.status === 'succeeded') {
      const paid = Number(payment.amount), refunded = Number(sums.refunded);
      const bookingStatus = refunded >= paid - 0.005 ? 'refunded' :
        refunded > 0 ? 'partially_refunded' : 'paid';
      await conn.execute('UPDATE bookings SET payment_status = ? WHERE id = ?', [bookingStatus, current.booking_id]);
    }
    return {
      refund_id: refundId, status: nextStatus, gateway_status: gatewayStatus,
      provider_refund_id: current.provider_refund_id,
      checked_at: new Date().toISOString()
    };
  });
}

module.exports = { initiate, reconcile };

function validateSetting(key, value) {
  const definition = SETTING_DEFINITIONS[key];
  if (!definition) {
    throw AppError.badRequest("This setting key is not editable through the admin API.", undefined, "SETTING_NOT_EDITABLE");
  }

  if (definition.type === "boolean") {
    if (typeof value !== "boolean") {
      throw AppError.badRequest(`${key} must be a boolean.`);
    }
    return value;
  }

  if (definition.type === "integer" || definition.type === "number") {
    const validNumber = definition.type === "integer"
      ? Number.isInteger(value)
      : typeof value === "number" && Number.isFinite(value);
    if (!validNumber || value < definition.min || value > definition.max) {
      throw AppError.badRequest(
        `${key} must be a ${definition.type} between ${definition.min} and ${definition.max}.`,
      );
    }
    return value;
  }

  if (typeof value !== "string" || value.length > definition.max) {
    throw AppError.badRequest(`${key} must be a string of at most ${definition.max} characters.`);
  }

  const normalized = value.trim();
  if (!normalized && (key === "support_email" || key === "support_phone" || key === "support_url")) {
    return "";
  }
  if (!normalized) {
    throw AppError.badRequest(`${key} must not be empty.`);
  }

  if (definition.type === "email" && !/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(normalized)) {
    throw AppError.badRequest("support_email must be a valid email address.");
  }
  if (definition.type === "currency" && !/^[A-Z]{3}$/.test(normalized)) {
    throw AppError.badRequest("default_currency must be a three-letter uppercase currency code.");
  }
  if (definition.type === "url") {
    let parsed;
    try {
      parsed = new URL(normalized);
    } catch {
      throw AppError.badRequest("support_url must be an absolute HTTP or HTTPS URL.");
    }
    if (!["http:", "https:"].includes(parsed.protocol)) {
      throw AppError.badRequest("support_url must use HTTP or HTTPS.");
    }
  }
  return normalized;
}

           (actor_user_id, action, entity_type, entity_id, before_data, after_data)

          JSON.stringify({ key, value, reason: normalizedReason }),
        ],
      );
      updates.push({ key, value, category: SETTING_DEFINITIONS[key].category });
    }

}

async function history({ page, limit, key }) {
  const conditions = ["log.action = 'settings.update'", "log.entity_type = 'admin_setting'"];
  const params = [];
  if (key) {
    if (!Object.prototype.hasOwnProperty.call(SETTING_DEFINITIONS, key)) {
      throw AppError.badRequest("Unknown setting key.", undefined, "SETTING_NOT_EDITABLE");
    }
    conditions.push("JSON_UNQUOTE(JSON_EXTRACT(log.after_data, '$.key')) = ?");
    params.push(key);
  }
  const where = `WHERE ${conditions.join(" AND ")}`;
  const offset = (page - 1) * limit;
  const [[countRows], [historyRows]] = await Promise.all([
    db.pool.execute(`SELECT COUNT(*) AS total FROM audit_logs log ${where}`, params),
    db.pool.execute(
      `SELECT log.id, log.actor_user_id, actor.full_name AS actor_name,
              log.before_data, log.after_data, log.created_at
         FROM audit_logs log
         LEFT JOIN users actor ON actor.id = log.actor_user_id
        ${where}
        ORDER BY log.created_at DESC, log.id DESC
        LIMIT ? OFFSET ?`,
      [...params, limit, offset],
    ),
  ]);

  const total = Number(countRows[0].total);
  return {
    items: historyRows.map((row) => {
      const before = JSON.parse(row.before_data);
      const after = JSON.parse(row.after_data);
      return {
        id: row.id,
        key: after.key,
        category: SETTING_DEFINITIONS[after.key]?.category || null,
        previousValue: before.value,
        value: after.value,
        reason: after.reason,
        actor: row.actor_user_id
          ? { id: row.actor_user_id, name: row.actor_name }
          : null,
        createdAt: row.created_at,
      };
    }),
    pagination: paginationMeta({ page, limit, total }),
  };
}

const crypto = require('crypto');
const { pool, withTransaction } = require('../../config/db');
const env = require('../../config/env');
const { queueBookingEmail } = require('../notifications/booking-events');

const API_BASE = env.SSLCOMMERZ_SANDBOX
  ? 'https://sandbox.sslcommerz.com'
  : 'https://securepay.sslcommerz.com';

function problem(statusCode, message) {
  const error = new Error(message);
  error.statusCode = statusCode;
  return error;
}

function money(value) {
  const n = Number(value);
  if (!Number.isFinite(n)) return null;
  return n.toFixed(2);
}

function requiredConfig() {
  if (!env.SSLCOMMERZ_STORE_ID || !env.SSLCOMMERZ_STORE_PASSWORD ||
      !env.SSLCOMMERZ_IPN_URL ||
      !env.SSLCOMMERZ_SUCCESS_URL || !env.SSLCOMMERZ_FAIL_URL ||
      !env.SSLCOMMERZ_CANCEL_URL || !env.FRONTEND_URL) {
    throw problem(503, 'SSLCommerz is not fully configured on the server.');
  }
}

async function postForm(url, values) {
  const body = new URLSearchParams();
  for (const [key, value] of Object.entries(values)) {
    if (value !== undefined && value !== null) body.set(key, String(value));
  }
  const response = await fetch(url, {
    method: 'POST',
    headers: { 'content-type': 'application/x-www-form-urlencoded' },
    body,
    signal: AbortSignal.timeout(20000),
  });
  if (!response.ok) throw problem(502, `SSLCommerz API returned HTTP ${response.status}.`);
  return response.json();
}

async function initiate(userId, bookingId, idempotencyKey) {
  requiredConfig();
  if (typeof idempotencyKey !== 'string' || !idempotencyKey.trim() || idempotencyKey.trim().length > 100) {
    throw problem(400, 'Provide an idempotency_key of 1–100 characters.');
  }

  const attempt = await withTransaction(async (conn) => {
    const [rows] = await conn.execute(
      'SELECT * FROM bookings WHERE id = ? AND user_id = ? FOR UPDATE', [bookingId, userId]
    );
    if (!rows.length) throw problem(404, 'Booking not found.');
    const booking = rows[0];

    const [existingKey] = await conn.execute(
      'SELECT id, booking_id, provider, provider_payment_id, status, raw_event FROM payments WHERE idempotency_key = ? LIMIT 1',
      [idempotencyKey.trim()]
    );
    if (existingKey.length) {
      const old = existingKey[0];
      if (Number(old.booking_id) !== Number(bookingId) || old.provider !== 'sslcommerz') {
        throw problem(409, 'This idempotency key is already used by another payment.');
      }
      const meta = old.raw_event ? JSON.parse(old.raw_event) : {};
      if (meta.gatewayPageURL) return { reused: true, paymentId: old.id, gatewayPageURL: meta.gatewayPageURL };
      if (old.status === 'succeeded') throw problem(409, 'This booking payment is already completed.');
      throw problem(409, 'This payment attempt already exists but has no reusable checkout URL. Check payment history or contact support.');
    }

    if (!['pending_review', 'awaiting_payment'].includes(booking.booking_status)) {
      throw problem(409, 'This booking is not eligible for payment.');
    }
    if (['paid', 'not_required', 'partially_refunded', 'refunded'].includes(booking.payment_status)) {
      throw problem(409, 'This booking does not currently require another payment.');
    }
    if (booking.currency_code !== 'BDT') throw problem(409, 'SSLCommerz checkout is configured for BDT bookings only.');
    const amount = money(booking.total_amount);
    if (!amount || Number(amount) < 10 || Number(amount) > 500000) {
      throw problem(409, 'SSLCommerz transaction amount must be between BDT 10.00 and BDT 500,000.00.');
    }

    const [open] = await conn.execute(
      "SELECT id FROM payments WHERE booking_id = ? AND provider = 'sslcommerz' AND status IN ('initiated','pending') LIMIT 1 FOR UPDATE",
      [bookingId]
    );
    if (open.length) throw problem(409, 'An SSLCommerz payment attempt is already open for this booking. Reuse its original request key.');

    const tranId = `BB${Date.now()}${crypto.randomBytes(6).toString('hex').toUpperCase()}`;
    const [insert] = await conn.execute(
      `INSERT INTO payments (booking_id, provider, provider_payment_id, idempotency_key, amount, currency_code, status)
       VALUES (?, 'sslcommerz', ?, ?, ?, 'BDT', 'initiated')`,
      [bookingId, tranId, idempotencyKey.trim(), amount]
    );
    await conn.execute("UPDATE bookings SET payment_status = 'pending' WHERE id = ?", [bookingId]);
    return { paymentId: insert.insertId, tranId, amount, booking };
  });

  if (attempt.reused) return attempt;

  try {
    const b = attempt.booking;
    const result = await postForm(`${API_BASE}/gwprocess/v4/api.php`, {
      store_id: env.SSLCOMMERZ_STORE_ID,
      store_passwd: env.SSLCOMMERZ_STORE_PASSWORD,
      total_amount: attempt.amount,
      currency: 'BDT',
      tran_id: attempt.tranId,
      success_url: env.SSLCOMMERZ_SUCCESS_URL,
      fail_url: env.SSLCOMMERZ_FAIL_URL,
      cancel_url: env.SSLCOMMERZ_CANCEL_URL,
      ipn_url: env.SSLCOMMERZ_IPN_URL,
      cus_name: b.customer_name,
      cus_email: b.customer_email,
      cus_phone: b.customer_phone,
      cus_add1: 'Bangladesh',
      cus_city: 'Dhaka',
      cus_country: 'Bangladesh',
      shipping_method: 'NO',
      product_name: `Bengal Beyond booking ${b.booking_reference}`,
      product_category: 'Tour',
      product_profile: 'general',
      value_a: String(attempt.paymentId),
      value_b: String(b.id),
    });
    const checkoutUrl = result.GatewayPageURL || result.redirectGatewayURL;
    if (result.status !== 'SUCCESS' || !checkoutUrl || !/^https:\/\//i.test(checkoutUrl)) {
      throw problem(502, result.failedreason || 'SSLCommerz did not return a valid checkout URL.');
    }
    await pool.execute(
      'UPDATE payments SET status = ?, raw_event = ? WHERE id = ?',
      ['pending', JSON.stringify({ sessionkey: result.sessionkey, gatewayPageURL: checkoutUrl, sessionResponse: result }), attempt.paymentId]
    );
    return { payment_id: attempt.paymentId, tran_id: attempt.tranId, status: 'pending', gateway_url: checkoutUrl };
  } catch (error) {
    await pool.execute(
      "UPDATE payments SET status = 'failed', raw_event = ? WHERE id = ? AND status <> 'succeeded'",
      [JSON.stringify({ initiation_error: error.message }), attempt.paymentId]
    );
    await pool.execute(
      "UPDATE bookings SET payment_status = CASE WHEN EXISTS (SELECT 1 FROM payments p WHERE p.booking_id = bookings.id AND p.status = 'succeeded') THEN 'paid' ELSE 'failed' END WHERE id = ?",
      [bookingId]
    );
    throw error;
  }
}

async function validateAndApply(payload) {
  requiredConfig();
  const valId = String(payload.val_id || '').trim();
  if (!valId || valId.length > 100) throw problem(400, 'Missing SSLCommerz validation ID.');
  const url = new URL(`${API_BASE}/validator/api/validationserverAPI.php`);
  url.searchParams.set('val_id', valId);
  url.searchParams.set('store_id', env.SSLCOMMERZ_STORE_ID);
  url.searchParams.set('store_passwd', env.SSLCOMMERZ_STORE_PASSWORD);
  url.searchParams.set('format', 'json');
  const response = await fetch(url, { signal: AbortSignal.timeout(20000) });
  if (!response.ok) throw problem(502, 'Could not validate transaction with SSLCommerz.');
  const verified = await response.json();

  const gatewayStatus = String(verified.status || '').toUpperCase();
  if (!['VALID', 'VALIDATED', 'FAILED', 'CANCELLED'].includes(gatewayStatus)) {
    return { verified: false, gateway_status: verified.status || 'UNKNOWN' };
  }
  if (String(verified.currency || '').toUpperCase() !== 'BDT') throw problem(400, 'Validated transaction currency does not match.');

  return withTransaction(async (conn) => {
    const [rows] = await conn.execute(
      'SELECT p.*, b.booking_status FROM payments p JOIN bookings b ON b.id = p.booking_id WHERE p.provider = \'sslcommerz\' AND p.provider_payment_id = ? FOR UPDATE',
      [String(verified.tran_id || '')]
    );
    if (!rows.length) throw problem(404, 'No local payment matches this SSLCommerz transaction.');
    const payment = rows[0];
    if (money(verified.amount) !== money(payment.amount) || String(verified.currency || '').toUpperCase() !== payment.currency_code) {
      throw problem(409, 'Validated amount or currency does not match the local payment.');
    }
    if (gatewayStatus === 'FAILED' || gatewayStatus === 'CANCELLED') {
      if (payment.status !== 'succeeded') {
        await conn.execute('UPDATE payments SET status = ?, raw_event = ? WHERE id = ?',
          [gatewayStatus === 'FAILED' ? 'failed' : 'cancelled', JSON.stringify({ validation: verified }), payment.id]);
        await conn.execute(
          "UPDATE bookings SET payment_status = CASE WHEN EXISTS (SELECT 1 FROM payments p WHERE p.booking_id = bookings.id AND p.status = 'succeeded') THEN 'paid' ELSE ? END WHERE id = ?",
          [gatewayStatus === 'FAILED' ? 'failed' : 'unpaid', payment.booking_id]
        );
      }
      return { verified: true, payment_id: payment.id, gateway_status: gatewayStatus };
    }
    if (gatewayStatus === 'VALID' && String(verified.risk_level || '0') === '1') {
      await conn.execute(
        "UPDATE payments SET status = 'pending', raw_event = ? WHERE id = ?",
        [JSON.stringify({ validation: verified, risk_hold: true }), payment.id]
      );
      return { verified: true, risk_hold: true, payment_id: payment.id };
    }
    if (String(verified.status).toUpperCase() === 'VALID' || String(verified.status).toUpperCase() === 'VALIDATED') {
      const firstTime = payment.status !== 'succeeded';
      await conn.execute(
        "UPDATE payments SET status = 'succeeded', verified_at = UTC_TIMESTAMP(), raw_event = ? WHERE id = ?",
        [JSON.stringify({ validation: verified }), payment.id]
      );
      await conn.execute("UPDATE bookings SET payment_status = 'paid' WHERE id = ?", [payment.booking_id]);
      if (firstTime) {
        await queueBookingEmail(conn, payment.booking_id, 'payment_received', { amount: payment.amount });
      }
      // A payment must not bypass the operator's pending-review decision.
      if (payment.booking_status === 'awaiting_payment') {
        await conn.execute("UPDATE bookings SET booking_status = 'confirmed', confirmed_at = UTC_TIMESTAMP() WHERE id = ?", [payment.booking_id]);
        await queueBookingEmail(conn, payment.booking_id, 'booking_confirmed');
      }
      return { verified: true, payment_id: payment.id, booking_id: payment.booking_id, payment_status: 'paid' };
    }
    return { verified: false, gateway_status: verified.status };
  });
}

async function getOwnedPayment(userId, paymentId) {
  const [rows] = await pool.execute(
    `SELECT p.id, p.booking_id, b.booking_reference, p.provider, p.provider_payment_id,
            p.amount, p.currency_code, p.status, p.verified_at, p.created_at
       FROM payments p JOIN bookings b ON b.id = p.booking_id
      WHERE p.id = ? AND b.user_id = ?`, [paymentId, userId]
  );
  if (!rows.length) throw problem(404, 'Payment not found.');
  return rows[0];
}

module.exports = { initiate, validateAndApply, getOwnedPayment };

async function queueReplyEmail(conn, ticketId, messageId, message) {
  try {
    const [[t]] = await conn.execute(
      `SELECT t.ticket_reference, t.subject, t.user_id,
              COALESCE(u.email, t.guest_email) AS email, COALESCE(u.full_name, t.guest_name) AS name
         FROM support_tickets t LEFT JOIN users u ON u.id = t.user_id WHERE t.id = ?`, [ticketId]);
    if (!t?.email) return;
    await enqueue({
      userId: t.user_id, channel: "email", recipient: t.email, templateKey: "support_reply",
      payload: {
        name: t.name || "there", ticket_reference: t.ticket_reference, subject: t.subject, message,
        contact_url: `${env.FRONTEND_URL.split(",")[0].trim().replace(/\/+$/, "")}/contact`,
      },
      idempotencyKey: `support_reply:${messageId}`, conn,
    });
  } catch (error) {
    logger.error("Could not queue support reply email", { ticketId, error: error.message });
  }
}
async function addStaffMessage(ticketId, staffId, message, isInternal) {
  return withTransaction(async (conn) => {
    const [[ticket]] = await conn.execute("SELECT id, status FROM support_tickets WHERE id = ? FOR UPDATE", [ticketId]);
    if (!ticket) httpError("Support ticket not found", 404);
    const [ins] = await conn.execute(
      `INSERT INTO support_messages (ticket_id, sender_type, sender_user_id, message_text, is_internal)
       VALUES (?, 'staff', ?, ?, ?)`, [ticketId, staffId, message, isInternal ? 1 : 0]);
    const i = isInternal ? 1 : 0;
    await conn.execute(
      `UPDATE support_tickets SET last_message_at = NOW(),
        status = CASE WHEN ? = 1 THEN status ELSE 'waiting_customer' END,
        resolved_at = CASE WHEN ? = 1 THEN resolved_at ELSE NULL END,
        closed_at = CASE WHEN ? = 1 THEN closed_at ELSE NULL END
       WHERE id = ?`, [i, i, i, ticketId]);
    if (!isInternal) await queueReplyEmail(conn, ticketId, ins.insertId, message);
    return { ticketId, messageAdded: true, internal: Boolean(isInternal) };
  });
}
async function updateTicket(ticketId, patch) {
  return withTransaction(async (conn) => {
    const [[ticket]] = await conn.execute("SELECT id FROM support_tickets WHERE id = ? FOR UPDATE", [ticketId]);
    if (!ticket) httpError("Support ticket not found", 404);
    const sets = [];
    const params = [];
    if (patch.status) {
      sets.push("status = ?"); params.push(patch.status);
      sets.push("resolved_at = CASE WHEN ? = 'resolved' THEN NOW() ELSE NULL END"); params.push(patch.status);
      sets.push("closed_at = CASE WHEN ? = 'closed' THEN NOW() ELSE NULL END"); params.push(patch.status);
    }
    if (patch.priority) { sets.push("priority = ?"); params.push(patch.priority); }
    if (patch.assignedTo !== undefined) { sets.push("assigned_to = ?"); params.push(patch.assignedTo); }
    if (!sets.length) httpError("No valid ticket fields supplied", 400);
    params.push(ticketId);
    await conn.execute(`UPDATE support_tickets SET ${sets.join(", ")} WHERE id = ?`, params);
    return { ticketId, updated: true };
  });
}

const CATEGORIES = new Set(["general","booking","payment","refund","tour","account","technical","other"]);

function requiredText(value, max, field) {
  if (typeof value !== "string" || !value.trim() || value.trim().length > max) {
    const error = new Error(`${field} is required and must be at most ${max} characters`);
    error.statusCode = 400;
    throw error;
  }
  return value.trim();
}
function optionalText(value, max, field) {
  if (value == null || value === "") return null;
  return requiredText(value, max, field);
}
function email(value) {
  const v = requiredText(value, 190, "Email").toLowerCase();
  if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(v)) {
    const error = new Error("A valid email address is required");
    error.statusCode = 400;
    throw error;
  }
  return v;
}
function enumValue(value, allowed, field) {
  if (value == null) return undefined;
  if (!allowed.has(value)) {
    const error = new Error(`Invalid ${field}`);
    error.statusCode = 400;
    throw error;
  }
  return value;
}
function positiveId(value, field = "id") {
  const n = Number(value);
  if (!Number.isSafeInteger(n) || n < 1) {
    const error = new Error(`A valid ${field} is required`);
    error.statusCode = 400;
    throw error;
  }
  return n;
}

async function submit({ tourId, bookingId, userId, rating, reviewText }) {
  return withTransaction(async (conn) => {
    const [[booking]] = await conn.execute(
      `SELECT b.id, b.user_id, b.booking_status, bi.tour_id
       FROM bookings b JOIN booking_items bi ON bi.booking_id = b.id
       WHERE b.id = ? AND bi.tour_id = ? FOR UPDATE`, [bookingId, tourId]);
    if (!booking) err("Booking does not contain this tour", 404);
    if (booking.user_id !== userId) err("This booking does not belong to the authenticated customer", 403);
    if (booking.booking_status !== "completed") err("A review can be submitted only after the booking is completed", 409);
    const [[existing]] = await conn.execute(
      "SELECT id FROM tour_reviews WHERE booking_id = ? AND tour_id = ? LIMIT 1 FOR UPDATE",
      [bookingId, tourId]);
    if (existing) err("A review has already been submitted for this booking and tour", 409);
    const [[user]] = await conn.execute("SELECT full_name FROM users WHERE id = ?", [userId]);
    if (!user) err("Customer account not found", 404);
    const [result] = await conn.execute(
      `INSERT INTO tour_reviews

function buildWhere(f) {
  const clauses = ["t.status = 'published'", "t.deleted_at IS NULL"];
  const params = [];
  if (f.q) {
    const term = `%${f.q.replace(/[\\%_]/g, "\\$&")}%`;
    clauses.push("(t.title LIKE ? ESCAPE '\\\\' OR t.summary LIKE ? ESCAPE '\\\\' OR t.destination LIKE ? ESCAPE '\\\\')");
    params.push(term, term, term);
  }
  if (f.destinationId) { clauses.push("t.destination_id = ?"); params.push(f.destinationId); }
  if (f.destination) { clauses.push("t.destination LIKE ?"); params.push(`%${f.destination}%`); }
  if (f.categoryId) {
    clauses.push(`EXISTS (SELECT 1 FROM tour_category_assignments ca
      JOIN tour_categories c ON c.id = ca.category_id AND c.status = 'active'
      WHERE ca.tour_id = t.id AND ca.category_id = ?)`);
    params.push(f.categoryId);
  }
  if (f.tagId) {
    clauses.push(`EXISTS (SELECT 1 FROM tour_tag_assignments ta
      JOIN tour_tags tg ON tg.id = ta.tag_id AND tg.status = 'active'
      WHERE ta.tour_id = t.id AND ta.tag_id = ?)`);
    params.push(f.tagId);
  }
  if (f.minDuration !== undefined) { clauses.push("t.duration_days >= ?"); params.push(f.minDuration); }
  if (f.maxDuration !== undefined) { clauses.push("t.duration_days <= ?"); params.push(f.maxDuration); }
  if (f.minPrice !== undefined) { clauses.push("t.base_price >= ?"); params.push(f.minPrice); }
  if (f.maxPrice !== undefined) { clauses.push("t.base_price <= ?"); params.push(f.maxPrice); }
  const hasDepartureFilter = !!(f.departureFrom || f.departureTo || f.availableOnly || f.minSeats || f.guests || f.sort === "departure_soon");
  if (hasDepartureFilter) {
    // Past departures cannot be booked, so they never count as a match.
    const dep = ["d.tour_id = t.id", "d.status = 'scheduled'", "d.starts_at >= UTC_TIMESTAMP()"];
    if (f.departureFrom) { dep.push("d.starts_at >= ?"); params.push(`${f.departureFrom} 00:00:00`); }
    if (f.departureTo) { dep.push("d.starts_at < DATE_ADD(?, INTERVAL 1 DAY)"); params.push(`${f.departureTo} 00:00:00`); }
    if (f.minSeats) { dep.push("(CAST(d.seat_capacity AS SIGNED) - CAST(d.seats_reserved AS SIGNED)) >= ?"); params.push(f.minSeats); }
    else if (f.availableOnly) dep.push("(d.seat_capacity > d.seats_reserved)");
    if (f.guests) {
      dep.push(`GREATEST(
        CAST(d.seat_capacity AS SIGNED) - CAST(d.seats_reserved AS SIGNED) - CAST(COALESCE((
          SELECT SUM(h.quantity) FROM seat_holds h
          WHERE h.departure_id = d.id AND h.hold_status = 'active'
            AND h.expires_at > UTC_TIMESTAMP()
        ), 0) AS SIGNED),
        0
      ) >= ?`);
      params.push(f.guests);
    }
    clauses.push(`EXISTS (SELECT 1 FROM tour_departures d WHERE ${dep.join(" AND ")})`);
  }
  return { sql: clauses.join(" AND "), params };
}
function orderBy(sort) {
  switch (sort) {
    case "price_asc": return "t.base_price ASC, t.id DESC";
    case "price_desc": return "t.base_price DESC, t.id DESC";
    case "duration_asc": return "t.duration_days ASC, t.id DESC";
    case "duration_desc": return "t.duration_days DESC, t.id DESC";
    case "newest": return "t.created_at DESC, t.id DESC";
    case "departure_soon": return "next_departure ASC, t.id DESC";

}

       FROM tour_tag_assignments ta JOIN tour_tags tg ON tg.id = ta.tag_id

    WHERE ${sql}

const ALLOWED_SORTS = new Set(["recommended", "price_asc", "price_desc", "duration_asc", "duration_desc", "newest", "departure_soon"]);
function bad(message) { const e = new Error(message); e.statusCode = 400; throw e; }
function intQuery(value, name, { min = 1, max = 1000000000, optional = true } = {}) {
  if (value === undefined || value === null || value === "") {
    if (optional) return undefined;
    bad(`${name} is required`);
  }
  if (!/^\d+$/.test(String(value))) bad(`${name} must be a positive integer`);
  const n = Number(value);
  if (!Number.isSafeInteger(n) || n < min || n > max) bad(`${name} must be between ${min} and ${max}`);
  return n;
}
function dateQuery(value, name) {
  if (value === undefined || value === "") return undefined;
  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(value))) bad(`${name} must use YYYY-MM-DD`);
  const d = new Date(`${value}T00:00:00.000Z`);
  if (Number.isNaN(d.getTime()) || d.toISOString().slice(0, 10) !== value) bad(`${name} is not a valid date`);
  return value;
}
function moneyQuery(value, name) {
  if (value === undefined || value === "") return undefined;
  if (!/^\d{1,10}(?:\.\d{1,2})?$/.test(String(value))) bad(`${name} must be a non-negative amount with up to 2 decimals`);
  return String(value);
}
function parseSearchQuery(query) {
  const page = intQuery(query.page, "page", { min: 1, max: 100000, optional: true }) || 1;
  const limit = intQuery(query.limit, "limit", { min: 1, max: 50, optional: true }) || 12;
  const guests = intQuery(query.guests, "guests", { min: 1, max: 20 });
  const minDuration = intQuery(query.minDuration, "minDuration", { min: 1, max: 3650 });
  const maxDuration = intQuery(query.maxDuration, "maxDuration", { min: 1, max: 3650 });
  const minPrice = moneyQuery(query.minPrice, "minPrice");
  const maxPrice = moneyQuery(query.maxPrice, "maxPrice");
  if (minDuration && maxDuration && minDuration > maxDuration) bad("minDuration cannot exceed maxDuration");
  if (minPrice !== undefined && maxPrice !== undefined && Number(minPrice) > Number(maxPrice)) bad("minPrice cannot exceed maxPrice");
  const departureFrom = dateQuery(query.departureFrom, "departureFrom");
  const departureTo = dateQuery(query.departureTo, "departureTo");
  if (departureFrom && departureTo && departureFrom > departureTo) bad("departureFrom cannot be after departureTo");
  const minSeats = intQuery(query.minSeats, "minSeats", { min: 1, max: 100 });
  const sort = query.sort || "recommended";
  if (!ALLOWED_SORTS.has(sort)) bad(`sort must be one of: ${[...ALLOWED_SORTS].join(", ")}`);
  const q = String(query.q || "").trim();
  if (q.length > 120) bad("q must be at most 120 characters");
  const destination = String(query.destination || "").trim();
  if (destination.length > 160) bad("destination must be at most 160 characters");
  return {

const { pool, withTransaction } = require("../../config/db");

const tables = {
  categories: { table: "tour_categories", join: "tour_category_assignments", fk: "category_id" },
  tags: { table: "tour_tags", join: "tour_tag_assignments", fk: "tag_id" },
};
function config(kind) {
  const c = tables[kind];
  if (!c) { const e = new Error("Unknown taxonomy type"); e.statusCode = 400; throw e; }
  return c;
}
async function list(kind, { includeInactive = false, q = "" } = {}) {
  const c = config(kind);
  const where = [];
  const params = [];
  if (!includeInactive) where.push("status = 'active'");
  if (q) { where.push("(name LIKE ? OR slug LIKE ?)"); params.push(`%${q}%`, `%${q}%`); }
  const sql = `SELECT id, name, slug, description, status, created_at, updated_at
    FROM ${c.table} ${where.length ? "WHERE " + where.join(" AND ") : ""}
    ORDER BY name ASC`;
  const [rows] = await pool.execute(sql, params);
  return rows;
}
async function get(kind, id) {
  const c = config(kind);
  const [rows] = await pool.execute(
    `SELECT id, name, slug, description, status, created_at, updated_at FROM ${c.table} WHERE id = ?`,
    [id]
  );
  if (!rows.length) { const e = new Error("Taxonomy item not found"); e.statusCode = 404; throw e; }
  return rows[0];
}
async function create(kind, data) {
  const c = config(kind);
  const [result] = await pool.execute(
    `INSERT INTO ${c.table} (name, slug, description, status) VALUES (?, ?, ?, ?)`,
    [data.name.trim(), data.slug, data.description || null, data.status || "active"]
  );
  return get(kind, result.insertId);
}
async function update(kind, id, data) {
  const c = config(kind);
  await get(kind, id);
  const fields = [], values = [];
  for (const key of ["name", "slug", "description", "status"]) {
    if (data[key] !== undefined) {
      fields.push(`${key} = ?`);
      values.push(key === "name" ? data[key].trim() : data[key]);
    }
  }
  if (fields.length) {
    fields.push("updated_at = CURRENT_TIMESTAMP");
    await pool.execute(`UPDATE ${c.table} SET ${fields.join(", ")} WHERE id = ?`, [...values, id]);
  }
  return get(kind, id);
}
async function remove(kind, id) {
  const c = config(kind);
  await get(kind, id);
  const [[usage]] = await pool.execute(`SELECT COUNT(*) AS total FROM ${c.join} WHERE ${c.fk} = ?`, [id]);
  if (usage.total) {
    const e = new Error("Cannot delete an item assigned to tours; set it inactive instead");
    e.statusCode = 409; throw e;
  }
  await pool.execute(`DELETE FROM ${c.table} WHERE id = ?`, [id]);
}
async function assign(tourId, categoryIds = [], tagIds = []) {
  return withTransaction(async (conn) => {
    const [[tour]] = await conn.execute("SELECT id FROM tours WHERE id = ? AND deleted_at IS NULL FOR UPDATE", [tourId]);
    if (!tour) { const e = new Error("Tour not found"); e.statusCode = 404; throw e; }
    for (const [ids, table, fk] of [
      [categoryIds, "tour_categories", "category_id"], [tagIds, "tour_tags", "tag_id"]
    ]) {
      const unique = [...new Set(ids.map(Number))];
      if (unique.some((n) => !Number.isSafeInteger(n) || n < 1)) {
        const e = new Error("All taxonomy IDs must be positive integers"); e.statusCode = 400; throw e;
      }
      if (unique.length) {
        const [found] = await conn.query(
          `SELECT id FROM ${table} WHERE status = 'active' AND id IN (${unique.map(() => "?").join(",")})`,
          unique
        );
        if (found.length !== unique.length) {
          const e = new Error(`One or more ${table === "tour_tags" ? "tag" : "category"} IDs are invalid or inactive`);
          e.statusCode = 400; throw e;
        }
      }
      const join = table === "tour_tags" ? "tour_tag_assignments" : "tour_category_assignments";
      await conn.execute(`DELETE FROM ${join} WHERE tour_id = ?`, [tourId]);
      for (const id of unique) await conn.execute(
        `INSERT INTO ${join} (tour_id, ${fk}) VALUES (?, ?)`, [tourId, id]
      );
    }
    return { tourId, categoryIds: [...new Set(categoryIds.map(Number))], tagIds: [...new Set(tagIds.map(Number))] };
  });
}
async function tourTaxonomy(tourId) {
  const [[tour]] = await pool.execute("SELECT id FROM tours WHERE id = ? AND deleted_at IS NULL", [tourId]);
  if (!tour) { const e = new Error("Tour not found"); e.statusCode = 404; throw e; }
  const [categories] = await pool.execute(
    `SELECT c.id, c.name, c.slug FROM tour_categories c
     JOIN tour_category_assignments a ON a.category_id = c.id WHERE a.tour_id = ? ORDER BY c.name`, [tourId]);
  const [tags] = await pool.execute(
    `SELECT t.id, t.name, t.slug FROM tour_tags t
     JOIN tour_tag_assignments a ON a.tag_id = t.id WHERE a.tour_id = ? ORDER BY t.name`, [tourId]);
  return { categories, tags };
}

const slugPattern = /^[a-z0-9]+(?:-[a-z0-9]+)*$/;
function text(value, field, max, required = false) {
  if (required && (typeof value !== "string" || !value.trim())) {
    const e = new Error(`${field} is required`); e.statusCode = 400; throw e;
  }
  if (value !== undefined && value !== null &&
      (typeof value !== "string" || value.trim().length > max)) {
    const e = new Error(`${field} must be a string of at most ${max} characters`);
    e.statusCode = 400; throw e;
  }
}
function validateTaxonomy(body, { slugRequired = false } = {}) {
  text(body.name, "name", 120, true);
  text(body.slug, "slug", 140, slugRequired);
  text(body.description, "description", 1000);
  if (body.slug !== undefined && !slugPattern.test(body.slug)) {
    const e = new Error("slug must contain lowercase letters, numbers and hyphens");
    e.statusCode = 400; throw e;
  }
  if (body.status !== undefined && !["active", "inactive"].includes(body.status)) {
    const e = new Error("status must be active or inactive"); e.statusCode = 400; throw e;
  }
}
function positiveId(value, field = "id") {
  const n = Number(value);
  if (!Number.isSafeInteger(n) || n < 1) {
    const e = new Error(`${field} must be a positive integer`); e.statusCode = 400; throw e;
  }
  return n;
}

  return value.normalize("NFKD").toLowerCase().replace(/[\u0300-\u036f]/g, "")

function buildWhere(filters) {
  const clauses = ["t.status = 'published'", "t.deleted_at IS NULL"];
  const values = [];

    clauses.push("(t.title LIKE ? OR t.summary LIKE ? OR t.destination LIKE ?)");
    const term = `%${filters.q}%`;
    values.push(term, term, term);
  }
  if (filters.destination) { clauses.push("t.destination LIKE ?"); values.push(`%${filters.destination}%`); }
  if (filters.minDays !== undefined) { clauses.push("t.duration_days >= ?"); values.push(filters.minDays); }
  if (filters.maxDays !== undefined) { clauses.push("t.duration_days <= ?"); values.push(filters.maxDays); }
  if (filters.minPrice !== undefined) { clauses.push("t.base_price >= ?"); values.push(filters.minPrice); }
  if (filters.maxPrice !== undefined) { clauses.push("t.base_price <= ?"); values.push(filters.maxPrice); }

}

  const [rows] = await pool.execute(
    `SELECT t.id, t.slug, t.title, t.summary, t.destination, t.duration_days,
            t.base_price, t.currency_code, t.cover_image_url, t.created_at, t.updated_at
       FROM tours t WHERE ${where.sql} ORDER BY ${orderBy} LIMIT ? OFFSET ?`,
    [...where.values, limit, offset],
  );
  const [[count]] = await pool.execute(`SELECT COUNT(*) AS total FROM tours t WHERE ${where.sql}`, where.values);
  return { rows, total: Number(count.total) };
}

async function findPublishedBySlug(slug) {
  const [rows] = await pool.execute(
    `SELECT t.id, t.slug, t.title, t.summary, t.description, t.destination,
            d.slug AS destination_slug,
            t.duration_days, t.base_price, t.currency_code, t.cover_image_url,
            t.cancellation_policy, t.created_at, t.updated_at,
            (SELECT GROUP_CONCAT(DISTINCT c.name ORDER BY c.name SEPARATOR ', ')
                FROM tour_category_assignments ca
                JOIN tour_categories c ON c.id = ca.category_id AND c.status = 'active'
               WHERE ca.tour_id = t.id) AS categories,
            (SELECT GROUP_CONCAT(DISTINCT tg.name ORDER BY tg.name SEPARATOR ', ')
                FROM tour_tag_assignments ta
                JOIN tour_tags tg ON tg.id = ta.tag_id AND tg.status = 'active'
               WHERE ta.tour_id = t.id) AS tags
       FROM tours t
       LEFT JOIN destinations d ON d.id = t.destination_id AND d.status = 'published'
       WHERE t.slug = ? AND t.status = 'published' AND t.deleted_at IS NULL LIMIT 1`,
    [slug],
  );
  return rows[0] || null;
}

module.exports = { list, findPublishedBySlug };

const { pool } = require("../../config/db");

async function list(userId, { page, limit, offset }) {
  const [items] = await pool.execute(
    `SELECT t.id AS tour_id, t.slug, t.title, t.summary, t.destination,
            t.duration_days, t.base_price, t.currency_code, t.cover_image_url,
            w.created_at AS added_at
       FROM wishlist_items w
       JOIN tours t ON t.id = w.tour_id
      WHERE w.user_id = ? AND t.deleted_at IS NULL
      ORDER BY w.created_at DESC, w.id DESC
      LIMIT ? OFFSET ?`,
    [userId, limit, offset]
  );
  const [[count]] = await pool.execute(
    `SELECT COUNT(*) AS total
       FROM wishlist_items w
       JOIN tours t ON t.id = w.tour_id
      WHERE w.user_id = ? AND t.deleted_at IS NULL`,
    [userId]
  );
  const total = Number(count.total);
  return {
    items,
    pagination: {
      page,
      limit,
      total,
      totalPages: Math.ceil(total / limit)
    }
  };
}

async function add(userId, tourId) {
  const [[tour]] = await pool.execute(
    "SELECT id FROM tours WHERE id = ? AND status = 'published' AND deleted_at IS NULL",
    [tourId]
  );
  if (!tour) {
    const error = new Error("Tour not found");
    error.statusCode = 404;
    throw error;
  }
  await pool.execute(
    `INSERT INTO wishlist_items (user_id, tour_id)
     VALUES (?, ?)
     ON DUPLICATE KEY UPDATE id = LAST_INSERT_ID(id)`,
    [userId, tourId]
  );
  return { tourId, added: true };
}

async function remove(userId, tourId) {
  const [result] = await pool.execute(
    "DELETE FROM wishlist_items WHERE user_id = ? AND tour_id = ?",
    [userId, tourId]
  );
  return { tourId, removed: result.affectedRows > 0 };
}

async function contains(userId, tourId) {
  const [[row]] = await pool.execute(
    "SELECT 1 AS saved FROM wishlist_items WHERE user_id = ? AND tour_id = ?",
    [userId, tourId]
  );
  return { tourId, contains: Boolean(row) };
}

module.exports = { list, add, remove, contains };

function positiveId(value, field = "id") {
  const n = Number(value);
  if (!Number.isSafeInteger(n) || n < 1) {
    const e = new Error(`${field} must be a positive integer`); e.statusCode = 400; throw e;
  }
  return n;
}
function pagination(query) {
  const page = query.page === undefined ? 1 : Number(query.page);
  const limit = query.limit === undefined ? 20 : Number(query.limit);
  if (!Number.isSafeInteger(page) || page < 1 || page > 100000) {
    const e = new Error("page must be a positive integer"); e.statusCode = 400; throw e;
  }
  if (!Number.isSafeInteger(limit) || limit < 1 || limit > 50) {
    const e = new Error("limit must be between 1 and 50"); e.statusCode = 400; throw e;
  }
  return { page, limit, offset: (page - 1) * limit };
}
module.exports = { positiveId, pagination };

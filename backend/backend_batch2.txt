/**
 * Optional API version header middleware.
 * Current routes remain backward-compatible at /api/*; clients may send
 * X-API-Version: 1. Unsupported versions are rejected explicitly.
 */
module.exports = function apiVersion(req, res, next) {
  const requested = req.get('X-API-Version');
  if (!requested) return next();
  if (requested !== '1') {
    return res.status(400).json({
      success: false,
      message: `Unsupported API version '${requested}'. Supported version: 1.`,
      requestId: req.requestId
    });
  }
  res.set('X-API-Version', '1');
  res.set('Vary', 'X-API-Version');
  return next();
};

const jwt = require("jsonwebtoken");
const { pool } = require("../config/db");
const env = require("../config/env");
const AppError = require("../utils/appError");
const asyncHandler = require("../utils/asynchandler");
const { recordSecurityEvent } = require("../modules/audit/audit-logger");

const ISSUER = "bengal-beyond-api";
const AUDIENCE = "bengal-beyond-client";

/**
 * Requires a valid Phase 2.1 access token and an active database account.
 * Sets req.auth = { userId, email, roles } and req.user to a safe user object.
 */
module.exports = asyncHandler(async function authenticate(req, res, next) {
  const header = req.get("authorization") || "";
  const match = /^Bearer\s+(.+)$/i.exec(header);
  if (!match) {
    await recordSecurityEvent({
      action: "auth.access.denied",
      req,
      details: { outcome: "denied", reason: "missing_access_token", method: req.method, path: req.path },
    });
    throw AppError.unauthorized("A valid access token is required.", "ACCESS_TOKEN_REQUIRED");
  }

  let payload;
  try {
    payload = jwt.verify(match[1], env.JWT_ACCESS_SECRET, {
      algorithms: ["HS256"],
      issuer: ISSUER,
      audience: AUDIENCE,
    });
  } catch {
    await recordSecurityEvent({
      action: "auth.access.denied",
      req,
      details: { outcome: "denied", reason: "invalid_access_token", method: req.method, path: req.path },
    });
    throw AppError.unauthorized("Access token is invalid or expired.", "INVALID_ACCESS_TOKEN");
  }

  if (
    !payload || payload.typ !== "access" ||
    typeof payload.sub !== "string" || !/^[1-9]\d*$/.test(payload.sub)
  ) {
    throw AppError.unauthorized("Access token is invalid or expired.", "INVALID_ACCESS_TOKEN");
  }

  const [rows] = await pool.execute(
    `SELECT u.id, u.email, u.full_name, u.account_status,
            r.role_key
       FROM users u
       LEFT JOIN user_roles ur ON ur.user_id = u.id
       LEFT JOIN roles r ON r.id = ur.role_id
      WHERE u.id = ?`,
    [payload.sub],
  );

  if (!rows.length || rows.some((row) => row.account_status !== "active")) {
    throw AppError.unauthorized("Session is no longer valid.", "INVALID_SESSION");
  }

  const first = rows[0];
  const roles = [...new Set(rows.map((row) => row.role_key).filter(Boolean))];
  req.auth = { userId: first.id, email: first.email, roles };
  req.user = {
    id: first.id,
    email: first.email,
    fullName: first.full_name,
    accountStatus: first.account_status,
    roles,
  };
  return next();
});

const { pool } = require("../config/db");
const AppError = require("../utils/appError");
const asyncHandler = require("../utils/asynchandler");
const { recordSecurityEvent } = require("../modules/audit/audit-logger");

/**
 * Usage: router.get("/admin", authenticate, authorize("dashboard.read"), handler)
 * Permissions are resolved from the database on every request so permission
 * changes and role removals take effect without waiting for a token to expire.
 */
function authorize(...requiredPermissions) {
  if (
    requiredPermissions.length === 0 ||
    requiredPermissions.some((permission) => typeof permission !== "string" || !permission.trim())
  ) {
    throw new TypeError("authorize() requires at least one non-empty permission key.");
  }

  return asyncHandler(async function authorizeRequest(req, res, next) {
    if (!req.auth || !req.auth.userId) {
      throw AppError.unauthorized("Authentication is required.", "AUTHENTICATION_REQUIRED");
    }

    const placeholders = requiredPermissions.map(() => "?").join(", ");
    const [rows] = await pool.execute(
      `SELECT DISTINCT p.permission_key
         FROM user_roles ur
         JOIN role_permissions rp ON rp.role_id = ur.role_id
         JOIN permissions p ON p.id = rp.permission_id
        WHERE ur.user_id = ?
          AND p.permission_key IN (${placeholders})`,
      [req.auth.userId, ...requiredPermissions],
    );

    const granted = new Set(rows.map((row) => row.permission_key));
    const missing = requiredPermissions.filter((permission) => !granted.has(permission));
    if (missing.length) {
      await recordSecurityEvent({
        action: "auth.permission.denied",
        actorUserId: req.auth.userId,
        req,
        details: { outcome: "denied", permissions: missing, method: req.method, path: req.path },
      });
      throw AppError.forbidden("You do not have permission to perform this action.", "INSUFFICIENT_PERMISSION");
    }

    req.auth.permissions = [...new Set([...(req.auth.permissions || []), ...requiredPermissions])];
    return next();
  });
}

module.exports = authorize;

const AppError = require("../utils/appError");
const env = require("../config/env");
const logger = require("../utils/logger");

// Turns known library/database errors into AppErrors with a safe message.
// Raw SQL text is never sent to the client.
function mapKnownError(error) {
  if (error instanceof AppError) return error;
  if (Array.isArray(error?.issues) && /Zod/.test(error?.name || "")) {
    return AppError.validation(
      error.issues.map((i) => {
        const path = (i.path || []).join(".");
        return { path, message: path ? `${path}: ${i.message}` : i.message };
      })
    );
  }

  // body-parser
  if (error.type === "entity.parse.failed") {
    return AppError.badRequest("Request body is not valid JSON.", undefined, "INVALID_JSON");
  }
  if (error.type === "entity.too.large") {
    return new AppError(413, "PAYLOAD_TOO_LARGE", "Request body is too large.");
  }
  if (Number.isInteger(error.statusCode) && error.statusCode >= 400 && error.statusCode <= 599) {
    return new AppError(error.statusCode, error.code || "REQUEST_ERROR", error.message);
  }

  // CHECK constraint failures. Match by error NUMBER, not name: the mysql2 driver
  // only knows MySQL's error table, so MariaDB's 4025 arrives with a wrong name.
  if (error.errno === 3819 /* MySQL 8 */ || error.errno === 4025 /* MariaDB */) {
    return new AppError(422, "CONSTRAINT_VIOLATION", "The values do not satisfy a data rule.");
  }

  // MySQL / MariaDB (mysql2)
  switch (error.code) {
    case "ER_DUP_ENTRY":
      return AppError.conflict("A record with these values already exists.", undefined, "DUPLICATE_ENTRY");
    case "ER_NO_REFERENCED_ROW_2":
      return AppError.badRequest("A referenced record does not exist.", undefined, "INVALID_REFERENCE");
    case "ER_ROW_IS_REFERENCED_2":
      return AppError.conflict("This record is still in use and cannot be removed.", undefined, "RECORD_IN_USE");
    case "ER_LOCK_DEADLOCK":
    case "ER_LOCK_WAIT_TIMEOUT":
      return new AppError(503, "TRY_AGAIN", "The system is busy. Please try again.");
    default:
      return null;
  }
}

function buildErrorBody(error, { requestId, isProduction }) {
  const mapped = mapKnownError(error);

  if (mapped) {
    const body = { code: mapped.code, message: mapped.message, requestId };
    if (mapped.details !== undefined) body.details = mapped.details;
    return { status: mapped.statusCode, body: { success: false, error: body } };
  }

  // Unknown error = a bug. Never leak its message in production.
  return {
    status: 500,
    body: {
      success: false,
      error: {
        code: "INTERNAL_ERROR",
        message: isProduction ? "Something went wrong." : error.message,
        requestId,
      },
    },
  };
}

// Express recognises an error handler by its 4 arguments. Do not remove `next`.
// eslint-disable-next-line no-unused-vars
function errorHandler(error, req, res, next) {
  if (res.headersSent) return next(error);

  const { status, body } = buildErrorBody(error, {
    requestId: req.id,
    isProduction: env.IS_PRODUCTION,
  });

  if (status >= 500) {
    logger.error("Unhandled request error", {
      requestId: req.id,
      method: req.method,
      path: req.originalUrl,
      errorCode: error.code,
      stack: error.stack,
    });
  }

  return res.status(status).json(body);
}

module.exports = { errorHandler, mapKnownError, buildErrorBody };

const crypto = require("crypto");

const SAFE_ID = /^[A-Za-z0-9._-]{8,100}$/;

// Gives every request an id, returned in the X-Request-Id header and in
// every error body, so "it failed" can be matched to one log line.
module.exports = function requestId(req, res, next) {
  const incoming = req.get("x-request-id");
  req.id = incoming && SAFE_ID.test(incoming) ? incoming : crypto.randomUUID();
  res.set("X-Request-Id", req.id);
  next();
};

// Small reusable request-size guard. Keep route-specific schema validation in each module.
module.exports = function requestValidation({ maxJsonBytes = "1mb" } = {}) {
  return function validateRequestSize(req, res, next) {
    const length = Number(req.headers["content-length"] || 0);
    const max = typeof maxJsonBytes === "number"
      ? maxJsonBytes
      : Number.parseInt(maxJsonBytes, 10) * (maxJsonBytes.toLowerCase().includes("mb") ? 1024 * 1024 : 1024);
    if (Number.isFinite(length) && length > max) {
      return res.status(413).json({ success: false, message: "Request body is too large" });
    }
    next();
  };
};

const { ZodError } = require("zod");
const AppError = require("../utils/appError");

/**
 * Validates req.body, req.query, and/or req.params against Zod schemas.
 * Parsed & coerced values replace the original request objects AND populate
 * req.validated so both patterns work seamlessly.
 */
function validate({ body, query, params } = {}) {
  return (req, _res, next) => {
    try {
      req.validated = req.validated || {};
      if (params) {
        req.params = params.parse(req.params);
        req.validated.params = req.params;
      }
      if (query) {
        req.query = query.parse(req.query);
        req.validated.query = req.query;
      }
      if (body) {
        req.body = body.parse(req.body);
        req.validated.body = req.body;
      }
      return next();
    } catch (err) {
      if (err instanceof ZodError) {
        const issues = err.issues.map((i) => ({
          field: i.path.join(".") || "root",
          message: i.message,
        }));
        const firstMsg = issues[0] ? `${issues[0].field}: ${issues[0].message}` : "Invalid request input.";
        const appErr = AppError.badRequest(firstMsg, "VALIDATION_ERROR");
        appErr.details = issues;
        return next(appErr);
      }
      return next(err);
    }
  };
}

module.exports = validate;

async function check(path, expectedStatus = 200) {
  const response = await fetch(`${base}${path}`, { headers: { Accept: "application/json" } });
  const body = await response.text();
  if (response.status !== expectedStatus) {
    throw new Error(`${path}: expected HTTP ${expectedStatus}, received ${response.status}. Body: ${body.slice(0, 300)}`);
  }
  console.log(`OK ${response.status} ${path}`);
}

})();

const { pool, closePool } = require("../config/db");

const required = {
  destinations: ["id", "slug", "name"],
  tour_categories: ["id"],
  tour_category_assignments: ["tour_id", "category_id"],
  tour_tags: ["id"],
  tour_tag_assignments: ["tour_id", "tag_id"],
  wishlist_items: ["id", "user_id", "tour_id"],
  coupons: ["id", "code", "discount_type", "discount_value", "status"],
  coupon_redemptions: ["id", "coupon_id", "booking_id", "user_id", "discount_amount", "status"],
  support_tickets: ["id", "ticket_reference", "user_id", "status"],
  support_messages: ["id", "ticket_id", "sender_type", "message_text", "is_internal"],
  notification_templates: ["id", "template_key", "channel", "body_template"],
  notification_jobs: ["id", "channel", "recipient", "status", "idempotency_key"],
  notification_attempts: ["id", "job_id", "attempt_number", "outcome"],
  payment_webhook_events: [
    "id", "provider", "event_key", "validation_id", "payload_json",
    "processing_status", "attempt_count", "next_retry_at"
  ],
  tours: ["destination_id"]
};

async function main() {
  let failed = false;
  try {
    const [[db]] = await pool.execute("SELECT DATABASE() AS name, VERSION() AS version");
    console.log(`Database: ${db.name} | MariaDB/MySQL: ${db.version}`);
    for (const [table, columns] of Object.entries(required)) {
      const [rows] = await pool.execute(
        `SELECT COLUMN_NAME FROM information_schema.COLUMNS
          WHERE TABLE_SCHEMA = DATABASE() AND TABLE_NAME = ?`, [table]);
      const actual = new Set(rows.map((row) => row.COLUMN_NAME));
      const missing = columns.filter((column) => !actual.has(column));
      if (!rows.length) {
        failed = true;
        console.error(`MISSING TABLE  ${table}`);
      } else if (missing.length) {
        failed = true;
        console.error(`MISSING COLUMNS ${table}: ${missing.join(", ")}`);
      } else {
        console.log(`OK             ${table}`);
      }
    }
    if (failed) process.exitCode = 1;
    else console.log("Phase 5 schema preflight passed.");
  } catch (error) {
    console.error("Schema preflight failed:", error.message);
    process.exitCode = 1;
  } finally {
    await closePool();
  }
}
main();

const { verifyConnection, sendMail } = require("../services/mailer");
const env = require("../config/env");

async function main() {
  const to = process.argv[2];
  if (!to) {
    console.error("Usage: npm run email:test -- you@example.com");
    process.exitCode = 1;
    return;
  }
  try {
    console.log(`Connecting to ${env.SMTP_HOST}:${env.SMTP_PORT} as ${env.SMTP_USER}...`);
    await verifyConnection();
    console.log("SMTP connection and credentials OK.");
    const { messageId } = await sendMail({
      to,
      subject: "Bengal Beyond: test email",
      text: "This is a test email from the Bengal Beyond backend.",
    });
    console.log(`Message sent: ${messageId}`);
    console.log("Check delivery at https://mailtrap.io/sending/email_logs");
  } catch (error) {
    console.error("Email test failed:", error.message);
    process.exitCode = 1;
  }
}
main();

"use client";
import { useState } from "react";
import Container from "@/components/ui/Container";
import StatusBadge from "@/components/ui/StatusBadge";
import ConfirmModal from "@/components/ui/ConfirmModal";
import { useToast } from "@/components/ui/ToastProvider";
import { Loading, ErrorBox, Empty } from "@/components/account/State";
import useFetch from "@/hooks/useFetch";
import api from "@/lib/axios";
import { money, titleCase } from "@/lib/format";
import { errMsg } from "@/lib/errors";

const EMPTY = { destination_id: "", name: "", type: "hotel", star_rating: "4", base_rate_per_night: "", currency_code: "BDT", status: "active" };
const inp = "mt-1 w-full rounded-xl border border-border bg-white px-3 py-2 text-sm outline-none focus:border-accent";

const toArray = (v) => (Array.isArray(v) ? v : Array.isArray(v?.items) ? v.items : []);

export default function AdminAccommodation() {
  const toast = useToast();
  const { data, error, loading, reload } = useFetch("/admin/accommodation");
  const { data: destinationsRaw } = useFetch("/admin/destinations?limit=100");
  const destinations = toArray(destinationsRaw);
  const rows = toArray(data);

  const [form, setForm] = useState(EMPTY);
  const [editing, setEditing] = useState(null);
  const [busy, setBusy] = useState(false);
  const [deleteTarget, setDeleteTarget] = useState(null);

  const submit = async (e) => {
    e.preventDefault();
    setBusy(true);
    try {
      const payload = {
        destination_id: form.destination_id ? Number(form.destination_id) : null,
        name: form.name.trim(),
        type: form.type,
        star_rating: form.star_rating ? Number(form.star_rating) : null,
        base_rate_per_night: Number(form.base_rate_per_night),
        currency_code: form.currency_code.trim().toUpperCase() || "BDT",
        status: form.status,
      };
      if (editing) {
        await api.patch(`/admin/accommodation/${editing}`, payload);
        toast.success("Accommodation updated.");
      } else {
        await api.post("/admin/accommodation", payload);
        toast.success("Accommodation added.");
      }
      setForm(EMPTY);
      setEditing(null);
      await reload();
    } catch (err) {
      toast.error(errMsg(err));
    } finally {
      setBusy(false);
    }
  };

  const edit = (a) => {
    setEditing(a.id);
    setForm({
      destination_id: a.destination_id || "",
      name: a.name || "",
      type: a.type || "hotel",
      star_rating: a.star_rating ?? "4",
      base_rate_per_night: a.base_rate_per_night ?? "",
      currency_code: a.currency_code || "BDT",
      status: a.status || "active",
    });
  };

  const remove = async (id) => {
    setBusy(true);
    try {
      await api.delete(`/admin/accommodation/${id}`);
      toast.success("Accommodation deleted.");
      await reload();
    } catch (err) {
      toast.error(errMsg(err));
    } finally {
      setBusy(false);
    }
  };

  return (
    <Container as="main" className="py-8">
      <div>
        <h1 className="text-3xl text-primary">Accommodation partners</h1>
        <p className="mt-1 text-sm text-muted">Manage partner hotels, resorts, houseboats, and eco-camps.</p>
      </div>

      <form onSubmit={submit} className="mt-6 grid gap-4 rounded-2xl border border-border bg-white p-5 sm:grid-cols-2 lg:grid-cols-4">
        <label className="text-xs font-semibold text-primary">
          Property name *
          <input required value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold text-primary">
          Destination
          <select value={form.destination_id} onChange={(e) => setForm({ ...form, destination_id: e.target.value })} className={inp}>
            <option value="">Select destination…</option>
            {destinations.map((d) => <option key={d.id} value={d.id}>{d.name}</option>)}
          </select>
        </label>

        <label className="text-xs font-semibold text-primary">
          Type
          <select value={form.type} onChange={(e) => setForm({ ...form, type: e.target.value })} className={inp}>
            <option value="hotel">Hotel</option>
            <option value="resort">Resort</option>
            <option value="eco_camp">Eco Camp</option>
            <option value="houseboat">Houseboat</option>
            <option value="guesthouse">Guesthouse</option>
          </select>
        </label>

        <label className="text-xs font-semibold text-primary">
          Star rating (1–5)
          <input type="number" min={1} max={5} value={form.star_rating} onChange={(e) => setForm({ ...form, star_rating: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold text-primary">
          Base rate / night *
          <input required type="number" min={0} step="0.01" value={form.base_rate_per_night} onChange={(e) => setForm({ ...form, base_rate_per_night: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold text-primary">
          Currency
          <input maxLength={3} value={form.currency_code} onChange={(e) => setForm({ ...form, currency_code: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold text-primary">
          Status
          <select value={form.status} onChange={(e) => setForm({ ...form, status: e.target.value })} className={inp}>
            <option value="active">Active</option>
            <option value="inactive">Inactive</option>
          </select>
        </label>

        <div className="flex items-end gap-2">
          <button disabled={busy} className="btn-gold rounded-full px-6 py-2 text-sm font-semibold disabled:opacity-50">
            {busy ? "Saving…" : editing ? "Update" : "Add property"}
          </button>
          {editing && (
            <button type="button" onClick={() => { setEditing(null); setForm(EMPTY); }} className="rounded-full border border-border px-4 py-2 text-sm">
              Cancel
            </button>
          )}
        </div>
      </form>

      <div className="mt-6 overflow-x-auto rounded-2xl border border-border bg-white">
        {loading ? (
          <div className="p-6"><Loading /></div>
        ) : error ? (
          <div className="p-6"><ErrorBox error={error} onRetry={reload} /></div>
        ) : !rows.length ? (
          <div className="p-6"><Empty>No accommodation partners added yet.</Empty></div>
        ) : (
          <table className="w-full text-left text-sm">
            <thead className="border-b border-border text-xs uppercase tracking-wider text-muted">
              <tr>
                <th className="px-5 py-3">Property</th>
                <th>Destination</th>
                <th>Type</th>
                <th>Rating</th>
                <th>Rate / night</th>
                <th>Status</th>
                <th className="pr-5 text-right">Actions</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-border">
              {rows.map((a) => (
                <tr key={a.id} className="hover:bg-primary/5">
                  <td className="px-5 py-3 font-semibold text-primary">{a.name}</td>
                  <td>{a.destination_name || "—"}</td>
                  <td>{titleCase(a.type)}</td>
                  <td>{a.star_rating ? `${a.star_rating} ★` : "—"}</td>
                  <td className="font-semibold text-primary">{money(a.base_rate_per_night, a.currency_code)}</td>
                  <td><StatusBadge value={a.status} /></td>
                  <td className="space-x-3 whitespace-nowrap pr-5 text-right text-xs font-semibold">
                    <button onClick={() => edit(a)} className="text-primary underline">Edit</button>
                    <button onClick={() => setDeleteTarget(a)} className="text-terracotta underline">Delete</button>
                  </td>
                </tr>
              ))}
            </tbody>
          </table>
        )}
      </div>

      <ConfirmModal
        open={Boolean(deleteTarget)}
        tone="danger"
        title="Delete accommodation partner?"
        description={deleteTarget ? `Delete "${deleteTarget.name}" permanently?` : ""}
        confirmLabel="Delete"
        busy={busy}
        onConfirm={async () => {
          const target = deleteTarget;
          setDeleteTarget(null);
          if (target) await remove(target.id);
        }}
        onCancel={() => setDeleteTarget(null)}
      />
    </Container>
  );
}

import { useState } from "react";

import useFetch from "@/hooks/useFetch";
import { formatDateTime } from "@/lib/format";

export default function AdminActivityLogs() {
  const [tab, setTab] = useState("");            // "" | "/security-events"
  const [q, setQ] = useState("");
  const [search, setSearch] = useState("");
  const [page, setPage] = useState(1);

      <h1 className="text-3xl text-primary">Activity logs</h1>
      <div className="mt-6 flex flex-wrap items-center gap-3">

        ))}
        <form onSubmit={(e) => { e.preventDefault(); setPage(1); setSearch(q.trim()); }} className="ml-auto flex gap-2">
          <input value={q} onChange={(e) => setQ(e.target.value)} placeholder="Action, user, request id" aria-label="Search" className="w-60 rounded-xl border border-border bg-white px-3 py-2 text-sm" />

        </form>
      </div>
      <div className="mt-6 overflow-x-auto rounded-2xl border border-border bg-white">
        {loading ? <div className="p-6"><Loading /></div> : error ? <div className="p-6"><ErrorBox error={error} onRetry={reload} /></div> :
          !data.length ? <div className="p-6"><Empty>No events.</Empty></div> : (
          <table className="w-full text-left text-sm">
            <thead className="border-b border-border text-xs uppercase tracking-wider text-muted"><tr><th className="px-5 py-3">When</th><th>Actor</th><th>Action</th><th>Entity</th><th>IP</th><th>Result</th></tr></thead>
            <tbody className="divide-y divide-border">
              {data.map((l) => {
                const outcome = l.after_data?.outcome;
                return (
                  <tr key={l.id}>
                    <td className="whitespace-nowrap px-5 py-3">{formatDateTime(l.created_at)}</td>
                    <td>{l.actor_name || <span className="text-muted">anonymous</span>}</td>
                    <td className="font-mono text-xs">{l.action}</td>
                    <td className="text-xs">{l.entity_type}{l.entity_id ? ` #${l.entity_id}` : ""}</td>
                    <td className="font-mono text-xs">{l.ip_address || "—"}</td>
                    <td>{outcome && <span className={`rounded-full px-2 py-0.5 text-xs font-semibold ${outcome === "success" ? "bg-accent/15 text-primary" : "bg-terracotta/15 text-terracotta"}`}>{outcome}</span>}</td>
                  </tr>
                );
              })}
            </tbody>
          </table>
        )}
      </div>
      {meta?.totalPages > 1 && (
        <div className="mt-6 flex items-center justify-center gap-4 text-sm">
          <button disabled={page <= 1} onClick={() => setPage(page - 1)} className="rounded-full border border-primary px-4 py-2 disabled:opacity-40">Previous</button>
          <span>Page {page} of {meta.totalPages}</span>
          <button disabled={page >= meta.totalPages} onClick={() => setPage(page + 1)} className="rounded-full border border-primary px-4 py-2 disabled:opacity-40">Next</button>
        </div>
      )}
    </Container>
  );
}

import { formatDateTime, titleCase } from "@/lib/format";
import { errMsg } from "@/lib/errors";

const toArray = (v) => (Array.isArray(v) ? v : Array.isArray(v?.items) ? v.items : []);

export default function AdminAssignments() {
  const toast = useToast();
  const { data: assignments, error, loading, reload } = useFetch("/admin/assignments");

    e.preventDefault();
    if (!form.guide_id && !form.vehicle_id) return toast.error("Select a guide and/or vehicle.");
    setBusy(true);
    try {
      await api.post("/admin/assignments", {
        entity_type: "booking",
        entity_id: Number(form.entity_id),
        guide_id: form.guide_id ? Number(form.guide_id) : null,
        vehicle_id: form.vehicle_id ? Number(form.vehicle_id) : null,
        role_label: form.role_label.trim() || null,
        notes: form.notes.trim() || null,
      });
      toast.success("Resource assigned.");
      setForm({ entity_id: "", guide_id: "", vehicle_id: "", role_label: "Lead Guide", notes: "" });
      await reload();
    } catch (err) {
      toast.error(errMsg(err));
    } finally {
      setBusy(false);
    }

    setBusy(true);
    try {
      await api.delete(`/admin/assignments/${id}`);

      setBusy(false);

      <form onSubmit={assign} className="mt-6 grid gap-4 rounded-2xl border border-border bg-white p-5 sm:grid-cols-2 lg:grid-cols-5">

"use client";
import { useState } from "react";
import Link from "next/link";
import Container from "@/components/ui/Container";
import StatusBadge from "@/components/ui/StatusBadge";
import { Loading, ErrorBox, Empty } from "@/components/account/State";
import useFetch from "@/hooks/useFetch";
import api from "@/lib/axios";
import { money, formatDateTime, titleCase, toLocalInput, fromLocalInput } from "@/lib/format";
import { errMsg } from "@/lib/errors";

const STATUSES = ["scheduled", "sold_out", "cancelled", "completed"];
const sel = "rounded-xl border border-border bg-white px-3 py-2 text-sm";

export default function AdminAvailability() {
  const { data: tours } = useFetch("/admin/tours?limit=100");
  const [f, setF] = useState({ tour_id: "", status: "", from: "", to: "" });
  const [applied, setApplied] = useState(f);
  const [page, setPage] = useState(1);
  const [editing, setEditing] = useState(null);
  const [busy, setBusy] = useState("");
  const [msg, setMsg] = useState({ type: "", text: "" });

  const qs = new URLSearchParams({ page, limit: 25 });
  for (const [k, v] of Object.entries(applied)) if (v) qs.set(k, v);
  const { data, meta, error, loading, reload } = useFetch(`/admin/departures?${qs}`);

  const openEdit = (d) =>
    setEditing({
      id: d.id,
      tour_title: d.tour_title,
      starts_at: toLocalInput(d.starts_at),
      ends_at: toLocalInput(d.ends_at),
      seat_capacity: d.seat_capacity,
      price_override: d.price_override ?? "",
    });

  const saveEdit = async (e) => {
    e.preventDefault();
    setBusy("save");
    setMsg({ type: "", text: "" });
    try {
      await api.patch(`/admin/departures/${editing.id}`, {
        starts_at: fromLocalInput(editing.starts_at),
        ends_at: editing.ends_at ? fromLocalInput(editing.ends_at) : null,
        seat_capacity: Number(editing.seat_capacity),
        price_override: editing.price_override === "" ? null : Number(editing.price_override),
      });
      setEditing(null);
      setMsg({ type: "ok", text: "Departure updated." });
      await reload();
    } catch (err) {
      setMsg({ type: "error", text: errMsg(err) });
    } finally {
      setBusy("");
    }
  };

  const changeStatus = async (d, status) => {
    const warn = status === "cancelled"
      ? `Cancel departure for "${d.tour_title}"? Existing bookings must be handled separately.`
      : `Mark this departure as completed?`;
    if (!window.confirm(warn)) return;
    setBusy(`${d.id}-${status}`);
    setMsg({ type: "", text: "" });
    try {
      await api.patch(`/admin/departures/${d.id}/status`, { status });
      setMsg({ type: "ok", text: `Marked as ${status}.` });
      await reload();
    } catch (err) {
      setMsg({ type: "error", text: errMsg(err) });
    } finally {
      setBusy("");
    }
  };

  return (
    <Container as="main" className="py-8">
      <div className="flex flex-wrap items-center justify-between gap-3">
        <div>
          <h1 className="text-3xl text-primary">Tour availability</h1>
          <p className="mt-1 text-sm text-muted">Global departure schedule, seat capacity, and pricing overrides across all tours.</p>
        </div>
        <Link href="/admin/tours" className="rounded-full border border-primary px-5 py-2 text-sm font-semibold text-primary">
          Manage tours
        </Link>
      </div>

      <form
        onSubmit={(e) => {
          e.preventDefault();
          setPage(1);
          setApplied(f);
        }}
        className="mt-6 flex flex-wrap items-end gap-3"
      >
        <select aria-label="Tour" value={f.tour_id} onChange={(e) => setF({ ...f, tour_id: e.target.value })} className={`${sel} max-w-xs`}>
          <option value="">All tours</option>
          {(tours || []).map((t) => (
            <option key={t.id} value={t.id}>{t.title}</option>
          ))}
        </select>

        <select aria-label="Status" value={f.status} onChange={(e) => setF({ ...f, status: e.target.value })} className={sel}>
          <option value="">All statuses</option>
          {STATUSES.map((s) => (
            <option key={s} value={s}>{titleCase(s)}</option>
          ))}
        </select>

        <label className="text-xs font-medium text-muted">
          From
          <input type="date" value={f.from} onChange={(e) => setF({ ...f, from: e.target.value })} className={`${sel} ml-2`} />
        </label>

        <label className="text-xs font-medium text-muted">
          To
          <input type="date" value={f.to} onChange={(e) => setF({ ...f, to: e.target.value })} className={`${sel} ml-2`} />
        </label>

        <button className="rounded-full bg-primary px-5 py-2 text-sm font-semibold text-white">Filter</button>
        {meta?.total != null && <span className="text-sm text-muted">{meta.total} departures</span>}
      </form>

      {editing && (

          <div className="flex gap-3">
            <button onClick={() => setPendingAction("approve")} disabled={!!busy} className="rounded-full bg-terracotta px-5 py-2 text-sm font-semibold text-white disabled:opacity-50">
              {busy === "approve" ? "…" : "Approve & cancel booking"}
            </button>
            <button onClick={() => setPendingAction("reject")} disabled={!!busy} className="rounded-full border border-primary px-5 py-2 text-sm font-semibold text-primary disabled:opacity-50">

  );

const inp = "mt-1 w-full rounded-xl border border-border bg-white px-3 py-2 text-sm";

function Editor({ role, perms, onSaved }) {
  const toast = useToast();
  const isNew = role === "new";
  const [sel, setSel] = useState(new Set(isNew ? [] : role.permissions));
  const [meta, setMeta] = useState({ key: "", name: "", description: "" });
  const [uid, setUid] = useState("");
  const [busy, setBusy] = useState(false);
  const [removeUserTarget, setRemoveUserTarget] = useState(null);
  const users = useFetch(isNew ? null : `/admin/roles/${role.id}/users?limit=100`);

  useEffect(() => { setSel(new Set(isNew ? [] : role.permissions)); }, [role, isNew]);
  const locked = !isNew && role.isSystem;

  const groups = perms.reduce((m, p) => ((m[p.permission_key.split(".")[0]] ||= []).push(p), m), {});
  const flip = (k) => setSel((s) => { const n = new Set(s); n.has(k) ? n.delete(k) : n.add(k); return n; });

  const run = async (fn, ok) => {
    setBusy(true);
    try {
      await fn();
      toast.success(ok);
      await onSaved();
      await users.reload?.();
    } catch (e) {
      toast.error(errMsg(e));
    } finally {
      setBusy(false);
    }
  };

  const save = () => run(() => isNew
    ? api.post("/admin/roles", { key: meta.key.trim(), name: meta.name.trim(), ...(meta.description && { description: meta.description }), permissions: [...sel] })
    : api.put(`/admin/roles/${role.id}/permissions`, { permissions: [...sel] }), "Role permissions saved.");

  return (
    <div className="space-y-6 rounded-2xl border border-border bg-white p-6">
      {isNew ? (
        <div className="grid gap-3 sm:grid-cols-3">
          <label className="text-sm font-medium">Key<input className={inp} value={meta.key} onChange={(e) => setMeta({ ...meta, key: e.target.value })} placeholder="finance_manager" pattern="[a-z][a-z0-9_]*" /></label>
          <label className="text-sm font-medium">Name<input className={inp} value={meta.name} onChange={(e) => setMeta({ ...meta, name: e.target.value })} /></label>
          <label className="text-sm font-medium">Description<input className={inp} value={meta.description} onChange={(e) => setMeta({ ...meta, description: e.target.value })} /></label>
        </div>
      ) : (
        <div><h2 className="text-xl text-primary">{role.name}</h2><p className="text-sm text-muted">{role.role_key}{role.description ? ` · ${role.description}` : ""}</p></div>
      )}
      {locked && <p className="rounded-xl bg-gold/15 p-3 text-sm">System roles can&apos;t be modified.</p>}

      <fieldset disabled={locked} className="grid gap-5 sm:grid-cols-2 lg:grid-cols-3">
        {Object.entries(groups).map(([g, list]) => (
          <div key={g}>
            <p className="text-xs font-bold uppercase tracking-widest text-muted">{g}</p>
            {list.map((p) => (
              <label key={p.permission_key} title={p.description} className="mt-1 flex items-center gap-2 text-sm">
                <input type="checkbox" checked={sel.has(p.permission_key)} onChange={() => flip(p.permission_key)} className="accent-[var(--accent)]" />{p.permission_key}
              </label>
            ))}
          </div>
        ))}
      </fieldset>
      {!locked && <button onClick={save} disabled={busy || (isNew && (!meta.key || !meta.name))} className="btn-gold rounded-full px-7 py-2.5 text-sm font-semibold disabled:opacity-50">{busy ? "Saving…" : isNew ? "Create role" : "Save permissions"}</button>}

              </li>
            ))}
            {!(users.data || []).length && <li className="py-2 text-muted">Nobody yet.</li>}

          <form onSubmit={(e) => { e.preventDefault(); run(() => api.put(`/admin/roles/${role.id}/users/${uid.trim()}`), "Staff assigned.").then(() => setUid("")); }} className="mt-3 flex gap-3">

        description={removeUserTarget ? `Remove ${removeUserTarget.full_name} from ${role.name}?` : ""}

import Container from "@/components/ui/Container";
import StatusBadge from "@/components/ui/StatusBadge";
import ConfirmModal from "@/components/ui/ConfirmModal";
import { useToast } from "@/components/ui/ToastProvider";
import { Loading, ErrorBox, Empty } from "@/components/account/State";
import useFetch from "@/hooks/useFetch";
import api from "@/lib/axios";
import { money, titleCase } from "@/lib/format";
import { errMsg } from "@/lib/errors";

const TYPES = ["sedan", "suv", "microbus", "tourist_coach", "boat", "other"];
const STATUSES = ["available", "in_service", "maintenance", "inactive"];
const inp = "mt-1 w-full rounded-xl border border-border bg-white px-3 py-2 text-sm";
const BLANK = {
  name: "",
  vehicle_type: "microbus",
  registration_number: "",
  seat_capacity: 10,
  driver_name: "",
  driver_phone: "",
  daily_rate: "0",
  status: "available",
  notes: "",
};

export default function AdminVehicles() {
  const toast = useToast();
  const { data, error, loading, reload } = useFetch("/admin/vehicles");
  const [form, setForm] = useState(BLANK);
  const [editingId, setEditingId] = useState(null);
  const [busy, setBusy] = useState(false);
  const [deleteTarget, setDeleteTarget] = useState(null);
  const [deleting, setDeleting] = useState(false);

  const startEdit = (v) => {
    setEditingId(v.id);
    setForm({
      name: v.name,
      vehicle_type: v.vehicle_type,
      registration_number: v.registration_number,
      seat_capacity: v.seat_capacity,
      driver_name: v.driver_name || "",
      driver_phone: v.driver_phone || "",
      daily_rate: String(v.daily_rate),
      status: v.status,
      notes: v.notes || "",
    });
  };

  const reset = () => {
    setEditingId(null);
    setForm(BLANK);
  };

  const save = async (e) => {
    e.preventDefault();
    setBusy(true);
    try {
      const payload = { ...form, seat_capacity: Number(form.seat_capacity), daily_rate: Number(form.daily_rate) };
      if (editingId) {
        await api.patch(`/admin/vehicles/${editingId}`, payload);
        toast.success("Vehicle updated.");
      } else {
        await api.post("/admin/vehicles", payload);
        toast.success("Vehicle added to fleet.");
      }
      reset();
      await reload();
    } catch (err) {
      toast.error(errMsg(err));
    } finally {
      setBusy(false);
    }
  };

  const confirmDelete = async () => {
    if (!deleteTarget) return;
    setDeleting(true);
    try {
      await api.delete(`/admin/vehicles/${deleteTarget.id}`);
      toast.success(`Deleted vehicle "${deleteTarget.name}".`);
      setDeleteTarget(null);
      await reload();
    } catch (err) {
      toast.error(errMsg(err));
    } finally {
      setDeleting(false);
    }
  };

  return (
    <Container as="main" className="py-8">
      <h1 className="text-3xl text-primary">Vehicles &amp; transport</h1>
      <p className="mt-1 text-sm text-muted">Manage tour fleet vehicles, tourist boats, seat capacities, and assigned drivers.</p>

      <form onSubmit={save} className="mt-6 grid gap-4 rounded-2xl border border-border bg-white p-6 sm:grid-cols-2 lg:grid-cols-4">
        <div className="sm:col-span-2 lg:col-span-4">
          <h2 className="text-lg text-primary">{editingId ? `Edit vehicle #${editingId}` : "Add fleet vehicle"}</h2>
        </div>

        <label className="text-xs font-semibold">
          Vehicle name / model *
          <input required placeholder="Toyota HiAce Grand Cabin" value={form.name} onChange={(e) => setForm({ ...form, name: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold">
          Type
          <select value={form.vehicle_type} onChange={(e) => setForm({ ...form, vehicle_type: e.target.value })} className={inp}>
            {TYPES.map((t) => <option key={t} value={t}>{titleCase(t)}</option>)}
          </select>
        </label>

        <label className="text-xs font-semibold">
          Registration number *
          <input required placeholder="DHAKA-METRO-CHA-11-2345" value={form.registration_number} onChange={(e) => setForm({ ...form, registration_number: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold">
          Seat capacity
          <input type="number" min={1} max={120} value={form.seat_capacity} onChange={(e) => setForm({ ...form, seat_capacity: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold">
          Driver name
          <input value={form.driver_name} onChange={(e) => setForm({ ...form, driver_name: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold">
          Driver phone
          <input value={form.driver_phone} onChange={(e) => setForm({ ...form, driver_phone: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold">
          Daily rate (BDT)
          <input type="number" min={0} step="0.01" value={form.daily_rate} onChange={(e) => setForm({ ...form, daily_rate: e.target.value })} className={inp} />
        </label>

        <label className="text-xs font-semibold">
          Status
          <select value={form.status} onChange={(e) => setForm({ ...form, status: e.target.value })} className={inp}>
            {STATUSES.map((s) => <option key={s} value={s}>{titleCase(s)}</option>)}
          </select>
        </label>

        <div className="flex gap-3 sm:col-span-2 lg:col-span-4">
          <button disabled={busy} className="btn-gold rounded-full px-6 py-2.5 text-sm font-semibold disabled:opacity-50">
            {busy ? "Saving…" : editingId ? "Update vehicle" : "Add vehicle"}
          </button>
          {editingId && (
            <button type="button" onClick={reset} className="text-sm font-semibold text-primary underline">Cancel</button>
          )}
        </div>
      </form>

      <div className="mt-6 overflow-x-auto rounded-2xl border border-border bg-white">
        {loading ? (
          <div className="p-6"><Loading /></div>
        ) : error ? (
          <div className="p-6"><ErrorBox error={error} onRetry={reload} /></div>
        ) : !data.length ? (
          <div className="p-6"><Empty>No vehicles registered yet.</Empty></div>
        ) : (
          <table className="w-full text-left text-sm">
            <thead className="border-b border-border text-xs uppercase tracking-wider text-muted">
              <tr>
                <th className="px-5 py-3">Vehicle</th>
                <th>Registration</th>
                <th>Seats</th>
                <th>Driver</th>
                <th>Daily Rate</th>
                <th>Status</th>
                <th className="pr-5 text-right">Actions</th>
              </tr>
            </thead>
            <tbody className="divide-y divide-border">
              {data.map((v) => (
                <tr key={v.id} className="hover:bg-primary/5">
                  <td className="px-5 py-3">
                    <span className="font-semibold text-primary">{v.name}</span>
                    <span className="block text-xs text-muted">{titleCase(v.vehicle_type)}</span>
                  </td>
                  <td className="font-mono text-xs">{v.registration_number}</td>
                  <td>{v.seat_capacity} pax</td>
                  <td>
                    {v.driver_name || <span className="text-muted">—</span>}
                    {v.driver_phone && <span className="block text-xs text-muted">{v.driver_phone}</span>}
                  </td>
                  <td>{money(v.daily_rate, "BDT")}</td>
                  <td><StatusBadge value={v.status} /></td>
                  <td className="space-x-3 whitespace-nowrap pr-5 text-right text-xs font-semibold">
                    <button onClick={() => startEdit(v)} className="text-primary underline">Edit</button>

"use client";
import { useState } from "react";

import { useToast } from "@/components/ui/ToastProvider";
import api from "@/lib/axios";
import { errMsg } from "@/lib/errors";

export default function RefundActions({ refund: r, provider, onDone }) {
  const toast = useToast();
  const [busy, setBusy] = useState("");
  const [confirmState, setConfirmState] = useState(null);
  const base = `/admin/refunds/${r.id}`;

  const exec = async (name, fn, okText) => {
    setBusy(name);
    try {
      await fn();
      toast.success(okText);
      await onDone();
    } catch (e) {
      toast.error(errMsg(e));
    } finally {
      setBusy("");
    }
  };

  const ask = (name, title, description, confirmLabel, tone, fn, okText) => {
    setConfirmState({ title, description, confirmLabel, tone, run: () => exec(name, fn, okText) });
  };

  const b = "text-xs font-semibold underline disabled:opacity-40";
  if (!["requested", "pending"].includes(r.status)) return null;

  return (
    <div className="flex flex-wrap items-center gap-3">
      {r.status === "requested" && provider === "sslcommerz" && (
        <button
          className={`${b} text-primary`}
          disabled={!!busy}
          onClick={() =>
            ask(
              "submit",
              "Submit refund to SSLCommerz?",
              "Send this refund to SSLCommerz now? It cannot be sent twice.",
              "Submit to SSLCommerz",
              "primary",
              () => api.post(`${base}/sslcommerz/submit`),
              "Refund submitted to SSLCommerz."
            )
          }
        >
          {busy === "submit" ? "Sending…" : "Submit to SSLCommerz"}
        </button>
      )}
      {r.status === "pending" && (
        <button
          className={`${b} text-primary`}
          disabled={!!busy}
          onClick={() => exec("check", () => api.post(`${base}/sslcommerz/reconcile`), "Refund status reconciled.")}
        >
          {busy === "check" ? "Checking…" : "Check status"}
        </button>
      )}
      <button
        className={`${b} text-primary`}
        disabled={!!busy}
        onClick={() =>
          ask(
            "done",
            "Mark as refunded manually?",
            "Mark as refunded WITHOUT asking the gateway? Only do this if you already refunded it in the SSLCommerz merchant panel.",
            "Mark refunded",
            "primary",
            () => api.patch(base, { status: "succeeded" }),
            "Refund marked as succeeded."
          )
        }
      >
        Mark refunded
      </button>
      {r.status === "requested" && (
        <button
          className={`${b} text-terracotta`}
          disabled={!!busy}
          onClick={() =>
            ask(
              "cancel",
              "Cancel refund request?",
              "Cancel this refund request?",
              "Cancel refund",
              "danger",
              () => api.patch(base, { status: "cancelled" }),
              "Refund request cancelled."
            )
          }
        >
          Cancel
        </button>
      )}

      <ConfirmModal
        open={Boolean(confirmState)}
        tone={confirmState?.tone || "primary"}
        title={confirmState?.title}
        description={confirmState?.description}
        confirmLabel={confirmState?.confirmLabel || "Confirm"}
        busy={Boolean(busy)}
        onConfirm={async () => {
          const fn = confirmState?.run;
          setConfirmState(null);
          if (fn) await fn();
        }}
        onCancel={() => setConfirmState(null)}
      />
    </div>
  );
}

"use client";
import { useState } from "react";
import StatusBadge from "@/components/ui/StatusBadge";
import ConfirmModal from "@/components/ui/ConfirmModal";
import { useToast } from "@/components/ui/ToastProvider";
import RefundActions from "./RefundActions";
import useFetch from "@/hooks/useFetch";
import api from "@/lib/axios";
import { money, formatDateTime } from "@/lib/format";
import { errMsg } from "@/lib/errors";

function PaymentRefunds({ payment: p, reloadBooking }) {
  const toast = useToast();
  const { data, reload } = useFetch(`/admin/refunds?payment_id=${p.id}&limit=100`);
  const [amount, setAmount] = useState("");
  const [reason, setReason] = useState("");
  const [busy, setBusy] = useState(false);
  const [confirmOpen, setConfirmOpen] = useState(false);

  const items = data?.items || [];
  const committed = items.filter((r) => ["requested", "pending", "succeeded"].includes(r.status)).reduce((s, r) => s + Number(r.amount), 0);
  const refundable = Math.max(0, Math.round((Number(p.amount) - committed) * 100) / 100);
  const refresh = async () => { await Promise.all([reload(), reloadBooking()]); };

  const submitRefund = async () => {
    setBusy(true);
    try {
      await api.post(`/admin/refunds/payments/${p.id}`, { amount: Number(amount).toFixed(2), ...(reason.trim() && { reason: reason.trim() }) });
      toast.success(`Refund of ${money(amount, p.currency_code)} created.`);
      setAmount("");
      setReason("");
      await refresh();
    } catch (err) {
      toast.error(errMsg(err));
    } finally {
      setBusy(false);
    }
  };

  return (
    <div className="rounded-xl border border-border p-4">
      <p className="text-sm font-semibold text-primary">{p.provider} · {p.provider_payment_id} · paid {money(p.amount, p.currency_code)}</p>
      {items.length > 0 && (
        <ul className="mt-3 divide-y divide-border text-sm">
          {items.map((r) => (
            <li key={r.id} className="flex flex-wrap items-start justify-between gap-3 py-2">
              <span>{formatDateTime(r.created_at)} · {r.reason || "No reason"}</span>
              <span className="flex flex-wrap items-start gap-3"><StatusBadge value={r.status} /><strong>{money(r.amount, r.currency_code)}</strong><RefundActions refund={r} provider={p.provider} onDone={refresh} /></span>
            </li>
          ))}
        </ul>
      )}
      {refundable > 0 ? (
        <form onSubmit={(e) => { e.preventDefault(); setConfirmOpen(true); }} className="mt-4 flex flex-wrap items-end gap-3">
          <label className="text-xs font-semibold">Amount (max {refundable})
            <input type="number" required min="0.01" max={refundable} step="0.01" value={amount} onChange={(e) => setAmount(e.target.value)} className="mt-1 block w-32 rounded-xl border border-border px-3 py-2 text-sm" />
          </label>
          <button type="button" onClick={() => setAmount(String(refundable))} className="pb-2 text-xs font-semibold text-primary underline">Full</button>
          <label className="flex-1 text-xs font-semibold">Reason
            <input maxLength={500} value={reason} onChange={(e) => setReason(e.target.value)} className="mt-1 block w-full min-w-[12rem] rounded-xl border border-border px-3 py-2 text-sm" />
          </label>
          <button disabled={busy} className="rounded-full bg-primary px-5 py-2 text-sm font-semibold text-white disabled:opacity-50">{busy ? "…" : "Create refund"}</button>
        </form>
      ) : <p className="mt-3 text-xs text-muted">Fully refunded or committed.</p>}

      <ConfirmModal
        open={confirmOpen}
        tone="danger"
        title="Create refund record?"
        description={`Create a refund of ${money(amount || 0, p.currency_code)} for payment #${p.id}?`}
        confirmLabel="Create refund"
        busy={busy}
        onConfirm={async () => {
          setConfirmOpen(false);
          await submitRefund();
        }}
        onCancel={() => setConfirmOpen(false)}
      />
    </div>
  );
}

export default function RefundPanel({ payments, bookingStatus, reload }) {
  const paid = payments.filter((p) => p.status === "succeeded");
  if (!paid.length) return null;
  return (
    <section className="rounded-2xl border border-border bg-white p-5">
      <h2 className="text-lg text-primary">Refunds</h2>
      {!["cancelled", "rejected", "expired"].includes(bookingStatus) && (
        <p className="mt-1 text-sm text-primary-dark">This booking is still active. Refunds normally follow a cancellation.</p>
      )}
      <div className="mt-3 space-y-4">{paid.map((p) => <PaymentRefunds key={p.id} payment={p} reloadBooking={reload} />)}</div>
    </section>
  );
}

"use client";
import { useRef, useState } from "react";
import Container from "@/components/ui/Container";
import ConfirmModal from "@/components/ui/ConfirmModal";
import Pagination from "@/components/ui/Pagination";
import { useToast } from "@/components/ui/ToastProvider";
import { Loading, ErrorBox, Empty } from "@/components/account/State";
import useFetch from "@/hooks/useFetch";
import api from "@/lib/axios";
import { errMsg } from "@/lib/errors";
import { slugify } from "@/lib/format";

const inp = "mt-1 w-full rounded-xl border border-border bg-white px-3 py-2 text-sm outline-none focus:border-accent focus:ring-2 focus:ring-accent/20 disabled:bg-background disabled:text-muted";
const initial = (cfg) => Object.fromEntries(cfg.fields.map((f) => [f.name, f.type === "checkbox" ? !!f.default : f.default ?? ""]));

function Input({ f, value, onChange, disabled }) {
  const c = { id: `f-${f.name}`, name: f.name, required: f.required, disabled, placeholder: f.placeholder };
  if (f.type === "textarea") return <textarea {...c} rows={f.rows || 4} maxLength={f.maxLength} className={inp} value={value} onChange={(e) => onChange(e.target.value)} />;
  if (f.type === "select") return (
    <select {...c} className={inp} value={value} onChange={(e) => onChange(e.target.value)}>
      {f.options.map(([v, l]) => <option key={v} value={v}>{l}</option>)}
    </select>
  );
  if (f.type === "checkbox") return <input {...c} type="checkbox" checked={!!value} onChange={(e) => onChange(e.target.checked)} className="h-4 w-4 accent-[var(--accent)]" />;
  return <input {...c} type={f.type || "text"} min={f.min} max={f.max} step={f.step} maxLength={f.maxLength} pattern={f.pattern}
    className={inp} value={value} onChange={(e) => onChange(e.target.value)} />;
}

function Form({ cfg, mode, row, onDone, onCancel }) {
  const toast = useToast();
  const [values, setValues] = useState(() => (mode === "edit" ? cfg.toForm(row) : initial(cfg)));
  const [msg, setMsg] = useState("");
  const [busy, setBusy] = useState(false);
  const touched = useRef(new Set());

  const set = (f, val) => {
    if (f.autoSlug) touched.current.add(f.name);
    setValues((v) => {
      const n = { ...v, [f.name]: val };
      for (const g of cfg.fields) if (g.autoSlug === f.name && mode === "new" && !touched.current.has(g.name)) n[g.name] = slugify(val);
      return n;
    });
  };

  const submit = async (e) => {
    e.preventDefault();
    const bad = cfg.validate?.(values, mode);
    if (bad) return setMsg(bad);
    setBusy(true); setMsg("");
    try {
      const payload = cfg.toPayload(values, mode);
      if (mode === "new") await api.post(cfg.endpoint, payload);
      else await api[cfg.updateMethod || "patch"](cfg.itemPath(row), payload);
      await onDone(mode === "new" ? `${cfg.singular} created.` : `${cfg.singular} updated.`);
    } catch (err) {
      const m = errMsg(err);
      setMsg(m);
      toast.error(m);
      setBusy(false);
    }
  };

  return (
    <form onSubmit={submit} className="space-y-5 rounded-3xl border border-border bg-white p-6">
      <div className="grid gap-4 sm:grid-cols-2">
        {cfg.fields.map((f) => (
          <div key={f.name} className={f.wide ? "sm:col-span-2" : ""}>
            {f.type === "checkbox" ? (
              <label className="flex items-center gap-2 pt-6 text-sm font-medium">
                <Input f={f} value={values[f.name]} onChange={(v) => set(f, v)} />{f.label}
              </label>
            ) : (
              <label className="block text-sm font-medium">{f.label}{f.required && <span className="text-terracotta"> *</span>}
                <Input f={f} value={values[f.name]} onChange={(v) => set(f, v)} disabled={mode === "edit" && f.readOnlyOnEdit} />
              </label>
            )}
            {f.hint && <p className="mt-1 text-xs text-muted">{f.hint}</p>}
          </div>
        ))}
      </div>
      {msg && <p role="alert" className="text-sm text-terracotta">{msg}</p>}
      <div className="flex gap-3">
        <button disabled={busy} className="btn-gold rounded-full px-7 py-2.5 text-sm font-semibold disabled:opacity-60">{busy ? "Saving…" : mode === "new" ? `Create ${cfg.singular}` : "Save changes"}</button>
        <button type="button" onClick={onCancel} className="text-sm font-semibold text-primary underline">Cancel</button>
      </div>
    </form>
  );
}

export default function ResourceManager({ cfg }) {
  const toast = useToast();
  const defaults = cfg.defaultFilters || {};
  const pageSize = cfg.pageSize || 20;
  const [filters, setFilters] = useState(defaults);
  const [applied, setApplied] = useState(defaults);
  const [page, setPage] = useState(1);
  const [view, setView] = useState(null);
  const [busy, setBusy] = useState("");
  const [confirmState, setConfirmState] = useState(null); // { title, description, confirmLabel, tone, run }

  const qs = new URLSearchParams({ page, limit: pageSize, ...(cfg.query || {}) });
  for (const [k, v] of Object.entries(applied)) if (v) qs.set(k, v);
  const { raw, error, loading, reload } = useFetch(`${cfg.endpoint}?${qs}`);
  const { rows = [], pages = 1, total } = raw ? cfg.read(raw) : {};

  const guarded = async (key, fn) => {
    setBusy(key);
    try {
      await fn();
    } catch (e) {
      toast.error(errMsg(e));
    } finally {
      setBusy("");
    }
  };

  const edit = (row) => guarded(`e${row.id}`, async () =>
    setView({ mode: "edit", row: cfg.loadRow ? await cfg.loadRow(row, api) : row }));

  const remove = (row) => {
    setConfirmState({
      title: `${cfg.remove.label} ${cfg.singular}?`,
      description: cfg.remove.confirm(row),
      confirmLabel: cfg.remove.label,
      tone: "danger",
      run: async () => {
        await guarded(`d${row.id}`, async () => {
          await api.delete(cfg.itemPath(row));
          toast.success(`${cfg.remove.label} completed.`);
          await reload();
        });
      },
    });
  };

  const act = (a, row) => {
    const exec = () => guarded(`${a.label}${row.id}`, async () => {
      await a.run(row, api);
      toast.success(`${a.label} completed.`);
      await reload();
    });
    if (a.confirm) {
      setConfirmState({
        title: `${a.label}?`,
        description: a.confirm(row),
        confirmLabel: a.label,
        tone: a.danger ? "danger" : "primary",
        run: exec,
      });
    } else {
      exec();
    }
  };

  if (view) {
    return (
      <Container as="main" className="max-w-4xl py-8">
        <button onClick={() => setView(null)} className="text-sm text-accent link-underline">← {cfg.title}</button>
        <h1 className="mb-6 mt-3 text-3xl text-primary">{view.mode === "new" ? `New ${cfg.singular}` : `Edit ${cfg.singular}`}</h1>
        <Form cfg={cfg} mode={view.mode} row={view.row}
          onDone={async (t) => { setView(null); toast.success(t); await reload(); }} onCancel={() => setView(null)} />
      </Container>
    );
  }

  return (

              <thead className="border-b border-border text-xs uppercase tracking-wider text-muted">

                        <button key={a.label} disabled={!!busy} onClick={() => act(a, r)} className={`underline disabled:opacity-40 ${a.danger ? "text-terracotta" : "text-primary"}`}>{a.label}</button>
                      ))}
                      <button disabled={!!busy} onClick={() => edit(r)} className="text-primary underline disabled:opacity-40">Edit</button>
                      {cfg.remove && <button disabled={!!busy} onClick={() => remove(r)} className="text-terracotta underline disabled:opacity-40">{cfg.remove.label}</button>}
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
            <Pagination
              page={page}
              totalPages={pages}
              totalItems={total ?? rows.length}
              pageSize={pageSize}
              onPageChange={setPage}
            />
          </>
        )}
      </div>

      <ConfirmModal
        open={Boolean(confirmState)}
        tone={confirmState?.tone || "danger"}
        title={confirmState?.title}
        description={confirmState?.description}
        confirmLabel={confirmState?.confirmLabel || "Confirm"}
        busy={Boolean(busy)}
        onConfirm={async () => {
          const fn = confirmState?.run;
          setConfirmState(null);
          if (fn) await fn();
        }}
        onCancel={() => setConfirmState(null)}
      />
    </Container>
  );
}

        }}

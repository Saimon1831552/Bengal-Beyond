import { formatDate } from "@/lib/format";

export const revalidate = 300;

const API_BASE =
  process.env.INTERNAL_API_URL ||
  process.env.NEXT_PUBLIC_API_URL ||
  "http://localhost:5000/api";

async function fetchCmsPage(slug) {
  try {
    const res = await fetch(`${API_BASE}/public/pages/${encodeURIComponent(slug)}`, {
      next: { revalidate: 300 },
    });
    if (!res.ok) return null;
    const json = await res.json();
    return json?.data || null;
  } catch {
    return null;
  }
}

export async function generateMetadata({ params }) {
  const { slug } = await params;
  const page = await fetchCmsPage(slug);
  if (!page) return {};
  return buildMetadata({
    title: page.seo_title || page.title,
    description: page.seo_description || page.body?.slice(0, 160),
    path: `/pages/${page.slug}`,
  });
}

"use client";

import { useEffect, useState } from "react";
import Link from "next/link";
import { useRouter, useSearchParams } from "next/navigation";
import { authService } from "@/services/auth.service";
import { useAuthStore } from "@/store/useAuthStore";
import { loadAccess, homeFor } from "@/lib/access";

export default function SignInForm() {
  const router = useRouter();
  const raw = useSearchParams().get("next");
  const safe = raw && raw.startsWith("/") && !raw.startsWith("//") ? raw : null;
  const status = useAuthStore((s) => s.status);
  const [email, setEmail] = useState("");
  const [password, setPassword] = useState("");
  const [error, setError] = useState("");
  const [busy, setBusy] = useState(false);

  useEffect(() => {
    if (status !== "authenticated") return;
    let live = true;
    loadAccess().then((a) => live && router.replace(safe || homeFor(a)));
    return () => { live = false; };
  }, [status, safe, router]);

  const submit = async (e) => {
    e.preventDefault();
    setBusy(true); setError("");
    try { await authService.login(email.trim(), password); } // the effect above redirects
    catch (err) { setError(err.response?.data?.error?.message || "Could not sign in. Please try again."); }
    finally { setBusy(false); }
  };

  const input = "mt-1 w-full rounded-xl border border-border bg-white px-4 py-3 outline-none focus:border-accent";
  return (
    <form onSubmit={submit} className="mx-auto mt-10 w-full max-w-md space-y-4 rounded-3xl border border-border bg-white p-8 shadow-sm">
      <h1 className="text-3xl text-primary">Sign in</h1>
      <label className="block text-sm font-medium">Email
        <input type="email" required autoComplete="email" className={input} value={email} onChange={(e) => setEmail(e.target.value)} />
      </label>
      <label className="block text-sm font-medium">Password
        <input type="password" required autoComplete="current-password" className={input} value={password} onChange={(e) => setPassword(e.target.value)} />
      </label>
      {error && <p role="alert" className="text-sm text-terracotta">{error}</p>}
      <button disabled={busy} className="btn-gold w-full rounded-full px-6 py-3 font-semibold disabled:opacity-60">
        {busy ? "Signing in…" : "Sign in"}
      </button>
      <div className="flex justify-between text-sm">
        <Link href="/forgot-password" className="text-primary link-underline">Forgot password?</Link>
        <Link href="/register" className="text-primary link-underline">Create account</Link>
      </div>
    </form>
  );
}

"use client";
import { useEffect, useState } from "react";
import Link from "next/link";
import { useRouter } from "next/navigation";
import Field from "@/components/ui/Field";
import AuthShell from "./AuthShell";
import { authService } from "@/services/auth.service";
import { useAuthStore } from "@/store/useAuthStore";
import { errMsg } from "@/lib/errors";

export default function RegisterForm() {
  const router = useRouter();
  const status = useAuthStore((s) => s.status);
  const [f, setF] = useState({ fullName: "", email: "", password: "", phone: "", nationality: "" });
  const [error, setError] = useState("");
  const [busy, setBusy] = useState(false);
  const set = (k) => (e) => setF({ ...f, [k]: e.target.value });

  useEffect(() => { if (status === "authenticated") router.replace("/dashboard"); }, [status, router]);

  const submit = async (e) => {
    e.preventDefault();
    setBusy(true); setError("");
    try {
      await authService.register({ ...f, email: f.email.trim() });
      router.replace("/dashboard");
    } catch (err) { setError(errMsg(err)); } finally { setBusy(false); }
  };

  return (
    <AuthShell title="Create account" subtitle="Book tours and manage your trips in one place."
      footer={<>Already registered? <Link href="/sign-in" className="font-semibold text-primary link-underline">Sign in</Link></>}>

        <Field label="Full name" required minLength={2} autoComplete="name" value={f.fullName} onChange={set("fullName")} />
        <Field label="Email" type="email" required autoComplete="email" value={f.email} onChange={set("email")} />
        <Field label="Password" type="password" required minLength={12} autoComplete="new-password" hint="At least 12 characters." value={f.password} onChange={set("password")} />
        <Field label="Phone (optional)" type="tel" autoComplete="tel" value={f.phone} onChange={set("phone")} />
        <Field label="Nationality (optional)" value={f.nationality} onChange={set("nationality")} />
        {error && <p role="alert" className="text-sm text-terracotta">{error}</p>}

      </form>

  );
}

"use client";
import { useState } from "react";
import Link from "next/link";
import Field from "@/components/ui/Field";
import AuthShell from "./AuthShell";
import api from "@/lib/axios";
import { errMsg } from "@/lib/errors";

export default function ForgotForm() {
  const [email, setEmail] = useState("");
  const [done, setDone] = useState(false);
  const [error, setError] = useState("");
  const [busy, setBusy] = useState(false);

  const submit = async (e) => {
    e.preventDefault();
    setBusy(true); setError("");
    try { await api.post("/auth/request-password-reset", { email: email.trim() }); setDone(true); }
    catch (err) { setError(errMsg(err)); } finally { setBusy(false); }
  };

  return (
    <AuthShell title="Forgot password" subtitle="We'll email you a reset link, valid for 30 minutes."
      footer={<Link href="/sign-in" className="font-semibold text-primary link-underline">Back to sign in</Link>}>

        </p>
      ) : (
        <form onSubmit={submit} className="space-y-4">
          <Field label="Email" type="email" required autoComplete="email" value={email} onChange={(e) => setEmail(e.target.value)} />
          {error && <p role="alert" className="text-sm text-terracotta">{error}</p>}

import AuthShell from "./AuthShell";
import api from "@/lib/axios";
import { errMsg } from "@/lib/errors";

export default function ResetForm() {
  const token = useSearchParams().get("token") || "";
  const [password, setPassword] = useState("");
  const [confirm, setConfirm] = useState("");
  const [done, setDone] = useState(false);
  const [error, setError] = useState("");
  const [busy, setBusy] = useState(false);

  const submit = async (e) => {
    e.preventDefault();
    if (password !== confirm) return setError("Passwords do not match.");
    setBusy(true); setError("");
    try { await api.post("/auth/reset-password", { token, password }); setDone(true); }
    catch (err) { setError(errMsg(err)); } finally { setBusy(false); }
  };

  if (!token) return <AuthShell title="Invalid link" footer={<Link href="/forgot-password" className="font-semibold text-primary">Request a new link</Link>}><p className="text-sm text-muted">This reset link is missing its token.</p></AuthShell>;

  return (
    <AuthShell title="Set a new password">
      {done ? (
        <div role="status" className="space-y-4 text-sm">
          <p className="rounded-xl bg-accent/10 p-4 text-primary">Password updated. All other devices were signed out.</p>
          <Link href="/sign-in" className="btn-gold inline-block rounded-full px-6 py-3 font-semibold">Sign in</Link>
        </div>
      ) : (
        <form onSubmit={submit} className="space-y-4">
          <Field label="New password" type="password" required minLength={12} autoComplete="new-password" hint="At least 12 characters." value={password} onChange={(e) => setPassword(e.target.value)} />
          <Field label="Confirm password" type="password" required minLength={12} autoComplete="new-password" value={confirm} onChange={(e) => setConfirm(e.target.value)} />
          {error && <p role="alert" className="text-sm text-terracotta">{error}</p>}
          <button disabled={busy} className="btn-gold w-full rounded-full px-6 py-3 font-semibold disabled:opacity-60">{busy ? "Saving…" : "Update password"}</button>
        </form>
      )}
    </AuthShell>
  );
}

"use client";
import { useEffect, useRef, useState } from "react";
import Link from "next/link";
import { useSearchParams } from "next/navigation";
import AuthShell from "./AuthShell";
import api from "@/lib/axios";
import { errMsg } from "@/lib/errors";

export default function VerifyEmail() {
  const token = useSearchParams().get("token");
  const [state, setState] = useState(token ? "loading" : "error");
  const [msg, setMsg] = useState(token ? "" : "This link is missing its token.");
  const started = useRef(false); // tokens are single-use; Strict Mode would run the effect twice

  useEffect(() => {
    if (!token || started.current) return;
    started.current = true;
    api.post("/auth/verify-email", { token })
      .then(() => setState("ok"))
      .catch((e) => { setState("error"); setMsg(errMsg(e)); });
  }, [token]);

  return (
    <AuthShell title="Email verification">
      {state === "loading" && <p role="status" className="text-sm text-muted">Verifying…</p>}
      {state === "ok" && <p role="status" className="rounded-xl bg-accent/10 p-4 text-sm text-primary">Your email is verified. Thank you!</p>}
      {state === "error" && <p role="alert" className="text-sm text-terracotta">{msg} You can request a new link from your <Link href="/dashboard" className="font-semibold underline">dashboard</Link>.</p>}
    </AuthShell>
  );
}

import { useEffect } from "react";
import { useRouter } from "next/navigation";
import { useAuthStore } from "@/store/useAuthStore";

export default function RequireAuth({ children }) {
  const status = useAuthStore((s) => s.status);
  const router = useRouter();

  useEffect(() => {
    if (status === "guest") {
      const here = window.location.pathname + window.location.search;
      router.replace(`/sign-in?next=${encodeURIComponent(here)}`);
    }
  }, [status, router]);

  if (status !== "authenticated") {
    return <div role="status" className="grid min-h-[60vh] place-items-center text-muted">Loading…</div>;
  }
  return children;
}

"use client";

import { useEffect } from "react";
import { useAuthStore } from "@/store/useAuthStore";
import { useAccessStore } from "@/store/useAccessStore";
import { loadAccess } from "@/lib/access";
import { refreshSession } from "@/lib/axios";
import { hasSessionHint, SESSION_KEY } from "@/lib/session";

export default function AuthProvider({ children }) {
  const authStatus = useAuthStore((s) => s.status);

  useEffect(() => {
    // Remove tokens left in localStorage by the old implementation
    try { localStorage.removeItem("token"); localStorage.removeItem("user"); } catch {}

    const { clearAuth } = useAuthStore.getState();
    if (hasSessionHint()) {
      refreshSession().catch(() => clearAuth()); // single-flight: safe under Strict Mode
    } else {
      clearAuth();
    }

    // Logged out in another tab -> log out here too
    const onStorage = (e) => {
      if (e.key === SESSION_KEY && e.newValue !== "1") useAuthStore.getState().clearAuth();
    };
    window.addEventListener("storage", onStorage);
    return () => window.removeEventListener("storage", onStorage);
  }, []);

  useEffect(() => {
    const acc = useAccessStore.getState();
    if (authStatus === "authenticated" && acc.status === "idle") loadAccess();
    if (authStatus === "guest") acc.reset();
  }, [authStatus]);

  return children;
}

import api, { refreshSession } from "../lib/axios";
import { useAuthStore } from "../store/useAuthStore";
import { setSessionHint, clearSessionHint } from "../lib/session";

const unwrap = (res) => res.data?.data ?? res.data;

function saveSession(session) {
  if (!session?.accessToken) throw new Error("Authentication response did not include an access token.");
  useAuthStore.getState().setAuth(session.user || null, session.accessToken);
  setSessionHint();
  return session;
}

export const authService = {
  login: async (email, password) => saveSession(unwrap(await api.post("/auth/login", { email, password }))),
  register: async (data) => saveSession(unwrap(await api.post("/auth/register", data))),
  refreshSession,
  logout: async () => {
    try {
      await api.post("/auth/logout");
    } finally {
      clearSessionHint();
      useAuthStore.getState().clearAuth();
    }
  },
};

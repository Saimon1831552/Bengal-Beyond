import { Fraunces, Manrope } from "next/font/google";
import "./globals.css";
import { siteConfig } from "@/config/site";
import JsonLd from "@/components/seo/JsonLd";
import { organizationJsonLd, websiteJsonLd } from "@/lib/jsonld";
import SmoothScroll from "@/components/SmoothScroll";
import Navbar from "@/components/layout/Navbar";
import Footer from "@/components/layout/Footer";
import MaintenanceBanner from "@/components/layout/MaintenanceBanner";
import AuthProvider from "@/components/providers/AuthProvider";
import { ToastProvider } from "@/components/ui/ToastProvider";
import CustomCursor from "@/components/fx/CustomCursor";
import MicroFX from "@/components/fx/MicroFX";
import { getSiteInfo } from "@/services/content";


const fraunces = Fraunces({ subsets: ["latin"], variable: "--font-fraunces", display: "swap", style: ["normal", "italic"] });
const manrope  = Manrope({ subsets: ["latin"], variable: "--font-manrope", display: "swap" });

export const metadata = {
  metadataBase: new URL(siteConfig.url),
  title: { default: `${siteConfig.name} | ${siteConfig.tagline}`, template: `%s | ${siteConfig.name}` },
  description: siteConfig.description,
  applicationName: siteConfig.name,
  keywords: siteConfig.keywords,
  alternates: { canonical: "/" },
  openGraph: {
    type: "website",
    siteName: siteConfig.name,
    locale: siteConfig.locale,
    url: siteConfig.url,
    title: `${siteConfig.name} | ${siteConfig.tagline}`,
    description: siteConfig.description,
  },
  twitter: { card: "summary_large_image", title: siteConfig.name, description: siteConfig.description },
  robots: {
    index: true,
    follow: true,
    googleBot: { index: true, follow: true, "max-image-preview": "large", "max-snippet": -1, "max-video-preview": -1 },
  },
  verification: {
    google: siteConfig.verification.google || undefined,
    other: siteConfig.verification.bing ? { "msvalidate.01": siteConfig.verification.bing } : undefined,
  },
  formatDetection: { telephone: false },
};

export const viewport = {
  themeColor: siteConfig.themeColor,
  width: "device-width",
  initialScale: 1,
};

export default async function RootLayout({ children }) {
  const info = await getSiteInfo();

  return (
    <html lang={siteConfig.language} className={`${fraunces.variable} ${manrope.variable}`}>
      <body className="min-h-full flex flex-col">
        <SmoothScroll>
          <AuthProvider>
            <ToastProvider>
              <a href="#main" className="skip-link">Skip to content</a>
              <Navbar />
              <div id="main" className="flex-1">{children}</div>
              <Footer info={info} />
              <MaintenanceBanner info={info} />
            </ToastProvider>
          </AuthProvider>
          <CustomCursor />
          <MicroFX />
          <JsonLd data={[organizationJsonLd(), websiteJsonLd()]} />
        </SmoothScroll>
      </body>
    </html>
  );
}

===== FILE: frontend/src/app/robots.js =====
import { siteConfig, privatePaths } from "@/config/site";

export default function robots() {
  const isProd = process.env.VERCEL_ENV ? process.env.VERCEL_ENV === "production" : process.env.NODE_ENV === "production";

  // Never let staging / preview deployments get indexed
  if (!isProd) return { rules: { userAgent: "*", disallow: "/" } };

  return {
    rules: [{ userAgent: "*", allow: "/", disallow: privatePaths.map((p) => `${p}/`).concat(privatePaths) }],
    sitemap: `${siteConfig.url}/sitemap.xml`,
    host: siteConfig.url,
  };
}


===== FILE: frontend/src/app/sitemap.js =====
import { absoluteUrl } from "@/lib/seo";
import { getDestinations, getTours, getPosts } from "@/services/content";

const BUILD_DATE = new Date();

export default async function sitemap() {
  const [destinations, tours, posts] = await Promise.all([getDestinations(), getTours(), getPosts()]);

  const staticRoutes = [
    { path: "/", priority: 1, changeFrequency: "weekly" },
    { path: "/destinations", priority: 0.9, changeFrequency: "weekly" },
    { path: "/tours", priority: 0.9, changeFrequency: "weekly" },
    { path: "/blog", priority: 0.7, changeFrequency: "weekly" },
    { path: "/about", priority: 0.5, changeFrequency: "yearly" },
    { path: "/contact", priority: 0.5, changeFrequency: "yearly" },
  ].map((r) => ({ url: absoluteUrl(r.path), lastModified: BUILD_DATE, changeFrequency: r.changeFrequency, priority: r.priority }));

  const entry = (base, item, priority) => {
    const date = item.updatedAt || item.publishedAt;
    return {
      url: absoluteUrl(`${base}/${item.slug}`),
      lastModified: date ? new Date(date) : undefined,
      changeFrequency: "monthly",
      priority,
    };
  };

  return [
    ...staticRoutes,
    ...destinations.map((d) => entry("/destinations", d, 0.8)),
    ...tours.map((t) => entry("/tours", t, 0.8)),
    ...posts.map((p) => entry("/blog", p, 0.6)),
  ];
}
